\section{Proof of Theorem~\ref{thm:intersect}}
\subsection{Right-handed boxes}
\label{sec:boxes}

Let $f\colon \mathbb{R}\Curves^+(\Sigma) \to \mathbb{R}$ satisfy symmetry, additive union, smoothing, stability and homogeneity.
Recall that since $f$ satisfies the symmetry condition, we have that $f([a])=f([a^{-1}])$ for any $a \in \pi_1(S,\ast)$.
Recall, also, that the fact that $f$ satisfies the smoothing condition for unoriented curves means $f$ is non-increasing under smoothing essential crossings for any of the two possible unoriented resolutions (the one coherent with the chosen orientation and the one which isn't).
We will fix (arbitrarily) a simple closed oriented curve~$[x]$. The endpoints of the lifts of $[x]$  are dense in $\partial_{\infty} \Sigma$, by minimality of the action of $\pi_1(S,*)$ by deck transformations (see Proposition~\ref{prop:fundminimal}). For convenience, we will work in $\pi_1(S,\ast)$ with a basepoint~$\ast$ on~$[x]$ and use $x$ for the corresponding element of~$\pi_1$.

\begin{definition}[dreamcatcher]
\label{def:righthandeddreamcatcher}
Let $a,b,c,d \in \pi_1(S,\ast)$ so that:

\begin{enumerate}
    \item $[a],[b],[c],[d]$ cross $x$ to the right.
    \item $a c^{-1} d b^{-1} = 1$.
    \item There exists $n_0$ so that for all $n \geq n_0$, the lift of the path $ca^{-1}bx^n a c^{-1}$ and the path $cx^n$, both based at $*$, have a right handed crossing. See Figure~\ref{fig:rhbox} for examples and non-examples of dreamcatchers. We distinguish three types of dreamcatchers: type 1 has no nested lifts; type 2 has one pair of nested lifts with the same orientation; and type 3 has two pairs of nested lifts of $x$, each pair with the same orientation.
\end{enumerate}
We will call the datum $(a,b,c,d)$ a \emph{right-handed dreamcatcher}.
\end{definition}

\begin{remark}
\begin{enumerate} 
\item The reader might wonder why in item (3) we require the condition ``for $n$ large enough''. It turns out that requiring the condition for all $n$ is too restrictive, and while it does happen for some configurations of lifts, such as type 1, in general it doesn't hold for the types of configurations of lifts that we are interested in. See the clarification in the proof of Proposition~\ref{prop:positivity} for more on this.
\item Item (3) in the definition of dreamcatcher is equivalent to the following statement: there exists $n_0$ so that for all $n \geq n_0$, the lift of the path $ca^{-1}ax^n a c^{-1}$ and the path $dx^n$, both based at $*$, are not linked. 
     To see this, note that, for any $n$, the startpoint $(dx^n)^-$ appears before $\lim_n (dx^n)^-$ in counter-clockwise order. By item (3), $\lim_n (ca^{-1}ax^n a c^{-1})^-$ appears before $\lim_n (dx^n)^-$, and so for large enough $n$, $(dx^n)^-$ appears before $(ca^{-1}ax^n a c^{-1})^-$. The other implication is similar.
     \end{enumerate}
\end{remark}

Given a right-handed dreamcatcher, $(a,b,c,d)$ and given a choice of lift of the geodesic $[x]$, we obtain lifts differing from each other by translating by $a,b,c,d$, respectively. For example, $\tilde{x},a\cdot \tilde{x}, b \tilde{x}, aC \cdot \tilde{x}$.
There are other configurations of lifts possible, such as $\tilde{x},a\cdot \tilde{x}, b \tilde{x}, c \cdot \tilde{x}, Ba \cdot \tilde{x}$.
Fixing a lift of the curve $[x]$, this yields configurations of lifts of Type 1, Type 2 and Type 3, where the lifts of $x$ are placed as shown in Figure~\ref{fig:rhbox} respectively.
Note how the lifts of $[x]$ are non-intersecting, since $[x]$ is simple, and these orientation of the lifts is always right-handed.
A choice of one endpoint for each such lift of $x$, determines a box of geodesics in $G^+(\Sigma)$.

The elements $a,b,c,d$ in Figure~\ref{fig:rhbox} can be seen as deck transformations relating the lifts of $x$. 


As we have pointed out above, a dreamcatcher only determines a box up to (left) action of $\pi_1(S)$ by isometries.
We will now introduce a different notation which will be more helpful to refer to  concrete boxes.  Fix once and for all a lift $\tilde{x}$ of $x$ and then pick four arbitrary elements $p_1,q_1,p_2,q_2 \in \pi_1(S,*)$ sending $\tilde{x}$ to four respective distinct lifts of $x$, as in Figure~\ref{fig:newmodel3}, with the property that $p_2x^{-},p_1x^{-}, q_1x^{+}, q_2x^{+}$ appear in this counter clockwise order.
We will denote this datum by $(p_1,p_2,q_1,q_2)$.

We can relate the notations for a dreamcatcher in Figure~\ref{fig:rhbox} and Figure~\ref{fig:newmodel3} by setting
\begin{align}
  a &= q_1p_1 & b &= q_2p_1\\
  c &= q_1p_2 & d &= q_2p_2.
  \label{eq:relationdatum}
\end{align}


Note that $a,b,c,d$ satisfy a relation, whereas $p_1,p_2,q_1,q_2$ don't.

\begin{definition}
Fix once and for all a lift $\tilde{x}$ of $[x]$. Every right handed dreamcatcher given by the datum $(p_1,p_2,q_1,q_2)$  determines uniquely a \emph{right-handed box}, a half-closed half-open box in $\partial_{\infty} \Sigma \times \partial_{\infty} \Sigma - \Delta$ given by
\[
B_{p_1,p_2,q_1,q_2} \coloneqq [p_1x^{-}, p_2x^{-}) \times [q_2x^{+}, q_1x^{+}).
\]
A different choice of lift $\tilde{x}'$ determines a different box $B_{p_1,p_2,q_1,q_2}'$, whose endpoints differ from the endpoints of $B_{p_1,p_2,q_1,q_2}$ by conjugation by some element in $\pi_1(S,*)$.

To recap, the datum $(p_1,p_2,q_1,q_2)$ determines a box $B_{p_1,p_2,q_1,q_2}$ and the dreamcatcher datum $(a,b,c,d)$ determines a $\pi_1(S)$ orbit of boxes that we will denote $[B_{p_1,p_2,q_1,q_2}]$ or $[B]_{a,b,c,d}$ depending on if we want to drive attention to a concrete representative of the orbit, or to the orbit itself, respectively. Here $(p_1,p_2,q_1,q_2)$ is related to $(a,b,c,d)$ by Equations~\ref{eq:relationdatum}.

We will denote by $\mathcal{B}$ the set of right-handed boxes, and by 
\[\mathcal{E}=\{ (p_1 x^-, p_2 x^-, q_2 x^+, q_1x^+) : (a,b,c,d) \mbox{ determined by Eq~\ref{eq:relationdatum} form a dreamcatcher } \}\]
the set of 4-tuples of endpoints of right-handed boxes.
\end{definition}

Note that because of our interest in right-handed boxes, we are restricting our attention to a subset of the lifts of $[x]$.
However, these are still enough to approximate any box.


\begin{proposition}
The family of right handed boxes $\mathcal{B}$ is dense in $G^+(\Sigma)$, in the following sense.
For any rectangle $R = [p,q) \times [r,s) \in G^+(\Sigma)$,
and any $\epsilon>0$, there exist $a,b,c,d \in \pi_1(S,*)$ so that for some $B \in [B]_{a,b,c,d}$, with $B=[p',q') \times [r',s')$, we have $p' \in B_{\epsilon}(p)$, $q' \in B_{\epsilon}(q)$, $r' \in B_{\epsilon}(r)$ and $s' \in B_{\epsilon}(s)$.
\label{prop:density}
\end{proposition}
\begin{proof}
   Consider a lift $\tilde{x}$ of $[x]$ with startpoint $x^-$ and endpoint $x^+$.
   Given a points $p,q,r,s \in \partial_{\infty} \tilde{\Sigma}$ in this clockwise order, and any $\epsilon>0$, we produce lifts $\tilde{x}_p,\tilde{x}_q,\tilde{x}_r,\tilde{x}_s$ of $[x]$ with endpoints within $\epsilon$ of $p,q,r,s$, respectively.
   For $r$ (and similarly for $s$), we proceed as follows. Given a lift of $\tilde{x}$, and $p$, consider some element $\gamma \in \pi_1(\Sigma)$ with $\gamma_- \in (x^-, x^+)$ and $\gamma^+$ within $\epsilon/4$ of $p$. For $n$ large, we consider the lift of $[x]$ given by $\gamma^n \cdot \tilde{x}$, where $\cdot$ denotes the left action of $\gamma$, i.e., if $x$ is the element of $\pi_1(S,*)$ with invariant axis $\tilde{x}$, $\gamma^n \cdot \tilde{x}$ denotes the invariant axis of $\gamma^n x \gamma^{-n}$. For $n$ large enough we see that $(\gamma^n \cdot \tilde{x})^+$ and  $(\gamma^n \cdot \tilde{x})^-$ are within $\epsilon$ of $p$.   See Figure~\ref{fig:approx}.
   We will write $\gamma_r$ as $\gamma$ in this construction, and then repeat the same construction for $s$ with an element $\gamma_s$.
   This will give the necessary lifts with the correct orientations for $r$ and $s$.

\subsection{Measures of boxes}

Fix $x$ and a lift $\tilde{x}$ once and for all.

\begin{proposition}
\label{prop:limitexists}
Let $B_{p_1,p_2,q_1,q_2}$ be a right-handed box so that $[B_{p_1,p_2,q_1,q_2}]=[B]_{a,b,c,d}$.
The following limit associated to $B_{p_1,p_2,q_1,q_2}$ exists:
\[
\mu_{f}(B_{p_1,p_2,q_1,q_2})=\frac{1}{2}\lim_{n \to \infty}f([bx^n]) + f([cx^n])-f([ax^n]) - f([dx^n]).
\]
\end{proposition}
Before checking if this limit exists, note that its definition only depends on the $\pi_1(S)$ orbit of $B_{p_1,p_2,q_1,q_2}$, in other words, it only depends on the choice of right-handed dreamcatcher $(a,b,c,d)$. Thus, the pre-measure is $\pi_1(S)$-invariant.

We now prove Proposition~\ref{prop:limitexists} by breaking it into several lemmas.

\subsubsection{Smoothing lemmas}

We will need some key lemmas relating several curves by smoothing essential crossings.

\begin{lemma}
  Suppose $x,a \in \pi_1(S,\ast)$ and $[x]$ is simple.
  If $[a]$ crosses $[x]$ to the right, then we have the following
  geometric reductions:
  \begin{align}
    \label{eq:cross-red1}[a x^n][x^m] &\reducesto [a x^{n+m}]\\
    \label{eq:cross-red3}[a x^n][a x^m] & \reducesto [x^{\abs{n-m}}].
  \end{align}
  \label{lem:smoothingskey}
\end{lemma}
\begin{proof}
  Equation~(\ref{eq:cross-red1}) follows since $[a]$ (and $[a x^n]$)
  cross~$[x]$ to the right. For Equation~(\ref{eq:cross-red3}), consider the union of $[ax^n]$ and $[ax^m]$, and the crossing marked in the figure below. 
   \[\mathcenter{\svgfignormal{smint_1}}
\]
    There are two possible choices of resolution. We choose the one which does not agree with the natural orientations of $[a]$ and $[x]$.
    Smooth $[ax^n]$ with $[ax^m]$, as shown in the figure below ($n=4,m=3$), to obtain $[y]$ homotopic to $[x^{|n-m|}]$.
  \[\mathcenter{\svgfignormal{smint_2}}
\]
  It remains to show that the crossing was essential. We justify this in the proof of Lemma~\ref{lem:join-split-geom}.
\end{proof}


\begin{lemma}\label{lem:join-split-geom}
  If $[a]$ and $[b]$ cross $[x]$ to the right, then there integers $k$ and $s\geq 0$
  so that, for all large enough $n,m$ and $r$ with either $s < r < m-n+k-s$ or $-s > r > m-n+k+s$,
  \begin{align}
    \label{eq:cross_a} [a x^n] [b x^m] &\reducesto [ax^{n+r}bx^{m-r}]\\
    \label{eq:cross_b} [a x^nb x^m] &\reducesto [ax^{n+r}][bx^{m-r}].
  \end{align}
  If $x$ is simple it suffices to take $s=0$. If $a=b$, $k$ can be taken to be $0$.
\end{lemma}

\begin{corollary}\label{cor:additivity-limit}
  Let $x$ be simple. If $a$ crosses $x$ to the right, then
  \[
    \lim_{n \to \infty} f([a x^n]) - f([x^n])
  \]
  exists.
\end{corollary}

\begin{proof}[of Corollary~\ref{cor:additivity-limit}]
  By Equation~(\ref{eq:cross_a}), with $k=0$ (since $a=b$) and $s=0$ ($x$ is simple), $m=t-1$ and $n=t+1$ for arbitrary $t$, and $r=0$, for $n$ large enough we get
  \[
    [a x^{t-1}] [a x^{t+1}] \reducesto [ax^{t}ax^{t}].
  \]
  Then, we apply Equation~(\ref{eq:cross_b}) with $m=t$ and $n=t$,  and $r=0$, to get
    \[
    [a x^t a x^t] \reducesto [ax^{t}][ax^{t}].
  \]
  Thus, by the smoothing and additivity property of $f$, we get
  \[
  f([ax^{n-1}]) + f([ax^{n+1}]) \geq 2 f([ax^n]).
  \]
  Hence, $f([a x^n])$ is a convex function
  of $n$, and thus so is $f([a x^n]) - f([x^n])$. By
  Equations~(\ref{eq:cross-red1}) and~(\ref{eq:cross-red3}) (both with
  $n=0$),
  \[
    -f([a]) \le f([a x^n]) - f([x^n]) \le f([a]).
  \]
  A convex function that is bounded as $n \to \infty$ has a
  well-defined limit.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{cor:additivity-limit}]
It follows immediately from Corollary~\ref{cor:additivity-limit}.
\end{proof}

\begin{remark}
Convexity of $f([ax^n])$ as a function of $n$ also plays a key role in the work of Feng Luo of characterization of intersection numbers for simple closed curves (see \cite[Corollary~4.4]{Luo98:IntersectionNumbers}).
\end{remark}
