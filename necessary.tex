\section{Intersection numbers satisfy smoothing}
\label{sec:necessary}
In this section we give a detailed proof that intersection numbers
satisfy the hypotheses of our theorem. The only non-trivial condition
to check is smoothing. In \cite[Section 4.1]{MGT21:Smoothings} we gave
a topological proof of this fact for curves, and then extend the
result for any geodesic current by invoking density of weighted closed
curves in the space of currents.
Here we give a different proof for an arbitrary current that works in the universal cover, since it will be useful to prove later that measured laminations satisfy max-smoothing.

\begin{proposition}
Given a geodesic current $\mu$, the curve functional $i(\mu, \cdot)$ satisfies the hypotheses of Theorem~\ref{thm:intersect}.
\label{prop:necessary}
\end{proposition}
\begin{proof}
We will check each hypothesis:
\begin{itemize}
\item (additivity) By definition (see \cite{Bonahon86:EndsHyperbolicManifolds}, intersection number is bilinear in both factors. In particular, additivity holds.
\item (stability) Observe that given a curve $C$ with hyperbolic lift $\gamma$, and $x \in \gamma$, the set of geodesics $G(x, \gamma^n x]$ can be partitioned into $\bigcup_{i=1}^{n-1} G(x, \gamma^{i} \cdot x]$.  Thus, by \cite[Proposition 3.14(ii)]{MZ19:PositivelyRatioed}, 
\[
i(\mu, \gamma^n) = \mu(G(x, \gamma^n x]) = \sum_{i=1}^{n-1} \mu(G(x, \gamma^{i} \cdot x]) = n\mu(G(x, \gamma x]) = n i(\mu, \gamma).
\]
where in the second inequality we used finite additivity of $\mu$, and in the third one, $\Gamma$-invariance.
\item (smoothing) Follows from Proposition \ref{prop:smoothing_nec} below.
\end{itemize}
\end{proof}




\begin{proposition}\label{prop:smoothing_nec}
Suppose $a,b \in \pi_1(S)$ are non-trivial elements with distinct fixed points on 
$\partial_{\infty} S$, and let $\mu$ be any geodesic current. Then:
\begin{enumerate}

\item If $a$ and $b$ $R$-cross, then
\[
\begin{aligned}
i(\mu,a)+i(\mu,b) &\ge i(\mu,ab),\\
i(\mu,a)+i(\mu,b) &\ge i(\mu,ab^{-1}).
\end{aligned}
\]

\item If $a$ and $b$ are $R$-parallel, then
\[
\begin{aligned}
i(\mu,ab) &\ge i(\mu,a)+i(\mu,b),\\
i(\mu,ab) &\ge i(\mu,ab^{-1}).
\end{aligned}
\]

\item If $a$ and $b$ are $R$-anti-parallel, then
\[
\begin{aligned}
i(\mu,ab^{-1}) &\ge i(\mu,a)+i(\mu,b),\\
i(\mu,ab^{-1}) &\ge i(\mu,ab).
\end{aligned}
\]

\end{enumerate}
\end{proposition}



\begin{proof}
\begin{enumerate}
\item Let $p$ be the intersection point between the hyperbolic axis $\ora{a}$ of $a$ and $\ora{b}$ of $b$.
Since $p \in \ora{a}$, we have $x=b^{-1} \cdot p \in \ora{b}$.
We have $p$ and $y= a \cdot p$ are in $\ora{a}$. Thus
\begin{align*}
i(\mu, ab) \leq & \mu(G(x, ab x]) \leq \mu(G(x, b x]) + \mu(G(bx, ab x]) = \\
&  = \mu(G(b^{-1} p, p]) + \mu(G(p, ap]) = i(\mu, b) + i(\mu, a).
\end{align*}
In the first inequality we used~\cite[Proposition 3.14(i)]{MZ19:PositivelyRatioed} , in the second, we used that 
$G(x, b x] \subset G(x, b x] \cup G(bx, ab x]$, and in the last equality we used the fact that $p$ is in the
axis of $a$ and of $b$, and~\cite[Proposition 3.14(ii)]{MZ19:PositivelyRatioed}.
A similar argument with $ab^{-1}$ gives the other inequality.
\item The inequalities can be checked directly using a similar analysis as in the previous part. However, we can give a much more direct prove, as follows. Since $i(\mu, \dot)$ is continuous by~\cite[Proposition~4.5]{Bonahon86:EndsHyperbolicManifolds} and homogeneous, it follows from Proposition~\ref{prop:cont_conn_disc} that $i(\mu, \cdot)$ satisfies connected smoothing, as we wanted to show.
\item This case is the same as the second case, by changing $b$ to $b^{-1}$.
\end{enumerate}
\end{proof}
