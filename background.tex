\section{Background}
\label{sec:backround}

\subsection{General measure theory}
\label{subsec:backround}
Let $X$ be a set.

\begin{definition}[Semi-ring, ring, $\sigma$-ring]
  A family $\mathcal{S}$ of subsets of a set $X$ is called a
  \emph{semiring} (of sets) if it contains the empty set and, for
  every $A,B \in \mathcal{S}$, we have
  $A \cap B \in \mathcal{S}$ and if $A \subset B$ the difference
  $B \backslash A$ is a disjoint union of finitely many sets
  in~$\mathcal{S}$. A family $\mathcal{R}$ of subsets of a set $X$ is
  called a \emph{ring} if it contains the empty set and for all
  $A,B \in \mathcal{R}$, the sets $A \cap B$, $A \cup B$ and
  $A \backslash B$ belong to~$\mathcal{R}$. If $\mathcal{R}$ is a
  ring, the \emph{$\sigma$-ring} of $\mathcal{R}$ is the set of
  countable unions of elements of~$\mathcal{R}$.
  \label{def:semiring}
\end{definition}

\begin{definition}[semialgebra, algebra, $\sigma$-algebra]
  If in the above definitions one assumes also that $X$ is an element of
  the family, one gets, respectively, the definitions of
  \emph{semialgebra}, \emph{algebra} and \emph{$\sigma$-algebra}.
\end{definition}

\begin{lemma}
  For any semiring $\mathcal{S}$, the collections of all finite unions
  of sets in $\mathcal{S}$ forms a ring $\mathcal{R}_{\mathcal{S}}$,
  the \emph{ring generated by $\mathcal{S}$}. Every set in
  $\mathcal{R}_{\mathcal{S}}$ is a finite union of pairwise disjoint
  sets in~$\mathcal{S}$.
\end{lemma}
\begin{proof}
See \cite[1.2.14~Lemma]{Bo07:MeasureTheory}.
\end{proof}

\begin{definition}
A real-valued set function $\mu$ defined on a class of sets $\mathcal{A}$ is called \emph{finitely additive} if 
\[
\mu\Bigl(\bigcup\nolimits_{i=1}^n A_i \Bigr) = \sum_{i=1}^n \mu(A_i)
\]
for all $n$ and all disjoint sets $A_1,\cdots,A_n \in \mathcal{A}$ such that $\bigcup_{i=1}^n A_i \in \mathcal{A}$.
We say $\mu$ is \emph{countably subadditive} if, for all pairwise
disjoint sets $A_i \in \mathcal{A}$ such that $\bigcup_{i=1}^{\infty}
A_i \in \mathcal{A}$,
\[
\mu\Bigl(\bigcup\nolimits_{i=1}^{\infty} A_i \Bigr) \leq \sum_{i=1}^{\infty} \mu(A_i).
\]
We say $\mu$ is \emph{countably additive} if the above inequality is
replaced by an equality, which is true if and only if $\mu$ is
finitely additive and countably subadditive (see
\cite[1.3.9~Proposition]{Bo07:MeasureTheory}).
\end{definition}

\begin{proposition}
A finitely additive function $\mu$ on a semiring $\mathcal{S}$ extends
uniquely to a finitely additive function on $\mathcal{R}_{S}$.
\label{prop:addsemitoring}
\end{proposition}
\begin{proof}
See \cite[1.3.10~Proposition]{Bo07:MeasureTheory}.
\end{proof}


\begin{definition}
\label{def:outermeasure}
A set function $\mu$ defined on the class of all subsets of a set $X$ and taking values in $[0,\infty]$ is called an \emph{outer measure} if 
\begin{enumerate}
    \item $\mu(\emptyset)=0$;
    \item $\mu(A) \leq \mu(B)$ whenever $A \subset B$, i.e., $\mu$ is
      monotone; and
    \item $\mu$ is countably subadditive. 
\end{enumerate}
A set $A \subset X$ is called \emph{$\mu$-measurable} if, for every $E
\subset X$, one has the equality
\[
\mu(E \cap A) + \mu(E \backslash A) = \mu(E)
\]
The class of all $\mu$-measurable sets is denoted by $\mathcal{M}_{\mu}$, and is an algebra \cite[Theorem~1.11.4]{Bo07:MeasureTheory}.
An outer measure is called \emph{regular} if, for every set $A \subset X$, there exist a $\mu$-measurable set $B$ so that $A \subset B$ and $\mu(A)=\mu(B)$.
\end{definition}

\begin{theorem}[Carathéodory extension theorem]
Given a ring~$\mathcal{R}$ on a set~$X$ and an additive function~$\mu$ on~$\mathcal{R}$.
For $A \subset X$, let 
\[
\mu^*(A) \coloneqq \inf \biggl\{ \sum_{i=1}^n \mu(A_n) \biggm\vert A_n \in \mathcal{R}, A \subset \bigcup\nolimits_{i=1}^{\infty} A_i\biggr\},
\]
where in the case of absence of such a sequence~$A_n$ we set $\mu^*(A)
\coloneqq \infty$. Then
\begin{enumerate}
    \item $\mu^*$ is an outer regular measure;
    \item all the sets in $\mathcal{R}$ are $\mu$-measurable;
\end{enumerate}
Moreover, if $\mu$ is countably additive, we have
\begin{enumerate}
\item $\mu^*$ coincides with $\mu$ on $\mathcal{R}$; and
    \item $\mu$ extends as a measure on $\mathcal{M}_{\mu}$.
\end{enumerate}
\label{thm:caratheodory}
\end{theorem}
\begin{proof}
See \cite[1.11.8~Theorem]{Bo07:MeasureTheory}.
\end{proof}

From now on, let $X$ be a topological space.

\begin{definition}[Borel $\sigma$-algebra, Borel measure]
The \emph{Borel $\sigma$-algebra} is the smallest $\sigma$-algebra containing all open sets in $X$. A \emph{Borel measure} is a measure on $\mathcal{B}(X)$.
\end{definition}

\begin{definition}[Support of a Borel measure]
Let $\mu$ be a Borel measure on $X$. The \emph{support} of $\mu$ is defined as the set of all points $x \in X$ for which every open neighbourhood $N_x$ of $x$ has positive measure. We will denote it $\supp(\mu)$.
\label{def:support}
\end{definition}

\begin{definition}[Atom of a measure]
Given a measure space $(X,\mu)$, and $A\subset X$ a measurable subset, we say $A$ is an atom of $\mu$ if the following two conditions are satisfied:
\begin{enumerate}
    \item $\mu(A) > 0$
    \item For any measurable set $B \subset A$ properly contained, $\mu(B)=0$.
\end{enumerate}
\label{def:atommeasure}
\end{definition}

\subsection{Radon measures}

\begin{definition}
Let $X$ be a locally compact and Hausdorff space. A Borel measure $\mu$ on $X$ is  is \emph{locally finite} if every point has a neighborhood $U$ with $\mu(U)< \infty$.
The measure $\mu$ is \emph{inner regular} if one has
\[
\mu(U) = \sup \{ \mu(K) : K \subset U \mbox{ compact } \}
\]
for every $U$ open.
For a Borel measure, being outer regular can be phrased as
\[
   \mu(B) = \inf \{ \mu(U) : U \supset B \mbox{ open } \}
\]
for every Borel set $B$. The measure $\mu$ is \emph{Radon} if it is locally finite, inner regular and outer regular.
\label{def:radon}
\end{definition}

The Riesz Representation theorem yields a characterization of Radon measures in terms of the associated functional on $C_c(X)$:

\begin{theorem}
  Let $X$ be a locally compact Hausdorff space. For any positive linear functional $F$ on $C_c(X)$ there is a unique Radon measure $\mu$ on $X$ so that
   \[
     F(\psi) = \int_X \psi(x)\mu(x)
  \]
  for all $\psi \in C_c(X)$, and a functional $F$ is called positive if $F(\psi) \geq 0$ for every function $\psi \in C_c(X) \geq 0$ and $x \in X$. The functional $F$ is not assumed to be continuous (a positive linear function on $C_c(X)$ is automatically continuous).
  \label{thm:Riesz}
\end{theorem}

On a general topological space $X$, not all Borel measures are Radon.
If $X$ is a metric space, the Radon property follows from the property of tightness

\begin{definition}
  A \emph{Borel measure} $\mu$ is \emph{tight} if, for every $\epsilon>0$, there exists a compact set $K_{\epsilon}$ so that $\mu(X \backslash K_{\epsilon}) < \epsilon$
\end{definition}

\begin{proposition}
Let $X$ be a metric space.
A tight Borel measure $\mu$ on $X$ is Radon.
\label{prop:tightborel}
\end{proposition}
\begin{proof}
  See \cite[Proposition~2.1.8]{Bogachev18:WeakConvergence}.
\end{proof}

However, tightness and Borel is not equivalent to Radon property: indeed, using the axiom of choice one can construct a separable metric space with a Borel probability measure which is not Radon (see \cite[Example~1.1.2]{Bogachev18:WeakConvergence}).
On complete metric spaces, there are examples of Borel measures which are not Radon. See \cite[Proposition~4.1.6]{Bogachev18:WeakConvergence}.
However, the following Theorem of Ulam will play an important role.

\begin{theorem}
On a complete separable metric space all Borel measures are Radon.
\label{thm:Ulam}
\end{theorem}
\begin{proof}
  See \cite[Theorem~2.1.9]{Bogachev18:WeakConvergence}.
\end{proof}

\begin{definition}
  By the \emph{space of measures $\mathcal{M}(X)$} on a topological space $X$ is the set of all positive Radon measures on $X$.
\end{definition}

We define the following topology in the space of measures.

\begin{definition} The weak$^*$-topology on $\mathcal{M}(X)$ is the weakest topology with respect to which the map
\begin{align*}
  \mathcal{M}(X) &\to \mathbb{R}\\
  \mu &\mapsto \int \psi \mu
\end{align*}
is continuous for all $\psi \in C_c(X)$.
\end{definition}

We suppose that $\mathcal{M}(X)$ is endowed with the weak$^*$-topology.
If $X$ is locally compact, Theorem \ref{thm:Riesz} implies $\mathcal{M}(X)$ is Hausdorff.
In fact, even more is true.

\begin{theorem}
Let $X$ be locally compact. If $X$ is second countable, then $\mathcal{M}(X)$ is metrizable and second countable.
\label{thm:spacemeasureswell}
\end{theorem}

In fact, $\mathcal{M}(X)$ has the Heine-Borel property.
A set $A \subset \mathcal{X}$ is \emph{bounded} whenever
\[
  \sup_{\mu \in H} \int \psi \mu < \infty
\]
for every $\psi \in C_c(X)$.

\begin{theorem}
Let $X$ be a locally compact space. A closed set $A \subset \mathcal{M}(X)$ is compact if and only if it is bounded.
\label{thm:heineborelproperty}
\end{theorem}

In particular, the set of bounded measures on a locally compact space is compact.

\subsection{Space of geodesics}

\begin{definition}[Boundary at infinity]
\label{def:boundaryinfty}
Endow $S$ with a complete
hyperbolic  metric~$g$; we denote the pair $(S,g)$ by~$\Sigma$. Then
we can consider the metric
universal covering $p \colon \wt{\Sigma} \to \Sigma$, with
$\wt{\Sigma}$ isometric to the hyperbolic plane.
Two quasi-geodesic rays $c,c' \colon [0,\infty) \to \wt{\Sigma}$ are
said to be \emph{asymptotic} if there exists a constant $K$ for which
$d(c(t),c'(t)) \leq K$ for all $t \geq 0$. We define
$\partial_{\infty} \Sigma$, the \emph{boundary at infinity}
of~$S$, to be the set of equivalence classes of
asymptotic quasi-geodesic rays. This boundary at infinity is
independent of the hyperbolic structure on~$S$ up to canonical
homeomorphism.
\end{definition}

\begin{definition}[Space of oriented geodesics]

\label{def:spacegeodesics}
Let $G(\Sigma)$ denote the \emph{space of oriented geodesics} in
$\wt{\Sigma}$, i.e.,
\[
G(\Sigma) \coloneqq\partial_\infty \Sigma  \times\partial_\infty\Sigma - \Delta.
\]
Since this is independent of the hyperbolic structure, we will also
write $G(S)$.
\end{definition}


\begin{definition}\label{def:currents}
We define $\Curr(S)$, the space of geodesic currents on $S$,
to be the
space of $\pi_1(S)$-invariant and flip invariant Radon measures on
$G(S)$.
\end{definition}

 \subsection{Convex functions on the integers}
 \label{subsec:convexity}

 We recall some basic results on convexity and existence of limits of
 functions on the integers. We start with an old result and then
 generalize to two variables (which should be standard, but we have
 not found older references) and to less-standard notions of convexity.
 Let $\lim_{n^{+}}f(n)$ denote $\lim_{n \to +\infty}f(n)$.

 \begin{lemma}
 Let $g \colon \mathbb{\mathbb{N}} \to \mathbb{R}$ be a convex and bounded function.
 Then $\lim_{n^+} g(n)$ exists.
 \label{lem:onevarconvexbounded}
 \end{lemma}
 \begin{proof}
   Consider $h(n) \coloneqq g(n+1) - g(n)$. Convexity of $g$ implies
   that $h$ is monotone increasing. Thus if $h(n_0) > 0$,
   then $g(n)$ increases monotonically without bound for $n > n_0$,
   contradicting boundedness. Therefore $h(n) \le 0$ for all $n$, so $g$ is
   monotonically decreasing, so has a limit.
\end{proof}

  \begin{lemma}\label{lem:monotone-double-limit}
  Let $(a_{n,m})_{n,m \in \mathbb{N}}$ be a double sequence that is
  bounded and monotone, in the sense that if $n' \leq n$ and
  $m' \leq m$ then
  $a_{n',m'} \leq a_{n,m}$. Then the following limits
  exist and are equal:
  \[
  \lim_{n^+}\lim_{m^+} a_{n,m} = \lim_{m^+}\lim_{n^+} a_{n,m} =
    \lim_{n^+, m^+} a_{n,m} = \sup_{n,m \ge 0} a_{n,m}.
  \]
  \end{lemma}
This lemma appears in Habil \cite[Thm.\ 4.2]{Hab16:DoubleSeq}, but we
include a proof for reference.

  \begin{proof}
    Let $V = \sup_{n,m} a_{n,m}$. For any $\epsilon > 0$, let
    $A_\epsilon \coloneqq \{(n,m) \mid a_{n,m} > V - \epsilon\}$. By
    definition of $\sup$, this set is non-empty; say $(n_0,m_0) \in
    A_\epsilon$. By monotonicity, $A_\epsilon$ contains all $(n,m)$
    with $n\ge n_0, m \ge m_0$. Since $\epsilon$ is arbitrary, it
    follows that $\lim_{n^+,m^+} a_{n,m} = V$. It also follows that
    for $n \ge n_0$, $\lim_{m^+} a_{n,m} > V-\epsilon$. (This limit
    exists by single-variable monotonicity.) Again we see that the
    double limits exist and are equal to $V$.
  \end{proof}

\begin{definition}
  A function $g \colon \bbN \to \bbR$ is \emph{$R$-convex}
  if for all $r \ge R$ and $n \in \bbN$ for which the inequality makes
  sense, we have $g(n-r) + g(n+r) \geq
  2 g(n)$. Similarly a
  function $g \colon \mathbb{N} \times \mathbb{N} \to \mathbb{R}$ is
  \emph{$(R,S)$-axis-convex} if, for all $n,m$, $g(\cdot, m)$ is
  $R$-convex and $g(n, \cdot)$ is $S$-convex. We say that $g$ is
  \emph{axis convex} if it is $(1,1)$-axis-convex.
\end{definition}
 
 \begin{lemma}
   Let $g \colon \mathbb{N} \times \mathbb{N} \to \mathbb{R}$ be axis
   convex and bounded. Then the following limits exist and are equal:
   \[\lim_{n^+,m^+} g(n,m) = \lim_{n^+} \lim_{m^+} g(n,m) = \lim_{m^+} \lim_{n^+} g(n,m).\]
 \end{lemma}
\begin{proof}
  As in Lemma~\ref{lem:onevarconvexbounded}, the function $h_1(n,m) =
  g(n+1,m) - g(n,m)$ is increasing as a function of $n$ by convexity
  and hence non-positive by boundedness of $g$. Similarly for
  $h_2(n,m) = g(n,m+1) - g(n,m)$. Thus $g$ is monotonically decreasing
  and the result follows by Lemma~\ref{lem:monotone-double-limit}.
\end{proof}

\begin{lemma}
  Let $g$ be a bounded, $R$-convex function
  $g \colon \bbN \to \bbR$. Then $\lim_{n^+} g(n)$ exists.
 \label{lem:limexistrmidconvex}
\end{lemma}
\begin{proof}
  For $0 \le k < R$, define $g_k(n) \coloneqq g(k + nR)$. Then each
  $g_k$ is convex and bounded, and thus by
  Lemma~\ref{lem:onevarconvexbounded} has a well-defined limit as
  $n \to \infty$. We must show these limits agree for different~$k$.
  By $R$-convexity,
  $g(k + nR) \le \frac{1}{2} \bigl(g((k-1) + (n-1)R) + g((k+1) +
  (n+1)R\bigr)$. Taking the limit, we see that
  $\lim_{n^+} g_k \le \frac{1}{2} (\lim_{n^+} g_{k-1} + \lim_{n^+}
  g_{k+1})$. The limits are thus a convex function on $\bbZ/R\bbZ$ and
  thus constant.
\end{proof}
 
\begin{lemma}
  Let $g \colon \mathbb{N} \times \mathbb{N} \to \mathbb{R}$ be a
  bounded, $(R,S)$-axis-convex function. Then the
  following limits exist and are equal:
  \[
    \lim_{n^+,m^+} g(n,m) = \lim_{n^+} \lim_{m^+} g(n,m) = \lim_{m^+}
    \lim_{n^+} g(n,m).
  \]
  \label{lem:limexistrmidconvextwovar}
\end{lemma}

\begin{proof}
  As in Lemma~\ref{lem:limexistrmidconvex}, for $0 \le k < R$ and $0
  \le \ell < R$, let $g_{k,\ell}(n,m) \coloneqq g(k+nR, \ell+mS)$.
  Then each $g_{k,\ell}$ is axis convex and thus has a well-defined limit
  as $n,m \to \infty$ (agreeing with the iterated limits), while the
  same averaging argument shows that the limits don't depend on
  $k,\ell$ and thus the overall limits exist and are equal.
\end{proof}

The following result is straightforward.

 \begin{lemma}
 Let $f_m \colon \mathbb{Z} \to \mathbb{R}$ be a sequence of convex functions converging to a  function
 $f \colon \mathbb{Z} \to \mathbb{R}$. Then $f$ is also convex.
 \label{lem:limitconvex}
 \end{lemma}

\begin{remark}
  We use the term \emph{axis convex} because the condition is not
  strong enough to deserve the term convex. For instance, the
  function $g(n,m) = -mn$ is axis convex by the above definition, but
  $g(n+1,m+1) + g(n-1,m-1) - 2g(n,m) = -2 < 0$.
\end{remark}

 \subsection{Geodesic currents}
\label{subsec:currents}

Geodesic currents are measures. There are several competing viewpoints on where one can see this measures supported on.
One viewpoint is as measures supported on the space of geodesics of a surface.
This viewpoint generalizes easily to hyperbolic groups.
However, the space of geodesics of $\Sigma$ is not well behaved (not Hausdorff), so one has to lift geodesics to the universal cover and work with $\pi_1(\Sigma)$-invariant measures.
This is the viewpoint we will be using.


\begin{definition}[Geodesic current]\label{def:currents-1}
We define $\GC^+(S)$, the space of oriented geodesic currents on $S$,
to be the
space of $\pi_1(S)$-invariant (positive) Radon measures on
$G^{+}(S)$.
\end{definition}


%\subsection{Oriented vs unoriented currents}

We will be mostly working on the setting of oriented geodesic
currents, but most of our natural examples use unoriented currents.


\begin{definition}[Unoriented geodesic currents]
\label{def:unorientedcurrents}
Let $\GC(\Sigma) \subset \GC^{+}(\Sigma)$ denote the space of \emph{unoriented geodesics currents} defined by
\[
\GC(\Sigma) = \{ \mu \in  \GC^{+}(\Sigma) \mid \sigma_*(\mu)=\mu \}
\]
where $\sigma_* : \mathcal{G}\mathcal{C}^{+}(\Sigma)
\to\mathcal{G}\mathcal{C}^{+}(\Sigma)$ is the map on currents induced
by the \emph{flip map} $\sigma: G^{+}(\Sigma) \to G^{+}(\Sigma)$
that switches the two factors in the definition of $G^+(\Sigma)$,
reversing the orientation of the geodesic.
\end{definition}

There is a map $\Pi : \GC^{+}(\Sigma) \to\GC^{+}(\Sigma)$ 
given by $\Pi(\mu)\coloneqq
\frac{1}{2}(\mu + \sigma_*(\mu))$ with image the subset of unoriented currents.

%\subsection{Properties of geodesic currents as measures}

The $\pi_1(S)$-invariance of geodesic currents highly constrains the atoms (see Definition~\ref{def:atommeasure}) they can have.

\begin{proposition}[{\cite[Lemma~8.2.8]{Martelli16:IntroGeoTop}}]
The orbit $\pi_1(S)\widetilde{\gamma}$ for $\widetilde{\gamma} \in
G^+(S)$ is discrete if and only if $p(\widetilde{\gamma})$ is a closed
geodesic.
\end{proposition}


This says that the only $0$-dimensional subsets with atoms are subsets consisting of lifts of closed geodesics.
Let a \emph{pencil} $P(x)$ be the subset of $G^+(S)$ of geodesics with one endpoint at $x \in \bdy\tilde{\Sigma}$.

\begin{proposition}[{\cite[Lemma~2.6]{dRMG22:Duals}}]
\label{prop:spikes}
Suppose a pencil $P(x)$ does not contain any element $\widetilde{\gamma}$ with a discrete orbit.
Then $\mu(P(x))=0$ for any geodesic current $\mu$.
\end{proposition}
