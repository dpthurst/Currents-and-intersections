\section{Construction of the geodesic current}
\label{sec:construct}



\subsection{Countable additivity}
\label{sec:countable-additivity}

We will prove countable sub-additivity of the function $\wh\mu_{f}$ by
proving approximation results. We first find controlled sequences of
decreasing nested RH boxes that we can show have measure going to zero.

\begin{definition}
\label{def:accordion}
Let the RH lift system $(p_1,q_1,p_2,q_2)$ determine a right-handed
box
$B = [s_1,s_2) \times [t_2,t_1)$, where $s_i = \ell_i^-$ and $t_i =
r_i^+$. Further assume that $\ora{r_2}$ is nested inside $\ora{r_1}$,
so the box is of type~2 or~3.

As in Figure~\ref{fig:vertical-accordion},
define the \emph{vertical accordion} $F(p_1, q_1, p_2,\vec{q_2})$ of
boxes based at $B$ to be
the sequence of boxes, indexed by $k$, with $p_1, p_2, q_1$ constant
and $q_2$ translated by powers of $r_1 = q_1 x Q_1$:
\[
  p_1^{(k)}\coloneqq p_1,\qquad
  p_2^{(k)}\coloneqq p_2,\qquad
  q_1^{(k)}\coloneqq q_1,\qquad
  q_2^{(k)} \coloneqq q_1 x^k Q_1 q_2.
\]
This defines boxes whose  height becomes smaller as $k$
grows, with the top endpoint fixed.

\begin{figure}
\centering{
\resizebox{120mm}{!}{\svgfignormal{v-accordion}}
\caption{A vertical accordion $F(p_1,q_1, p_2,\vec{q_2})$, shown in
    the disk (left) and in the whiskers picture (right)}
  \label{fig:vertical-accordion}
}
\end{figure}

Similarly, if $(\ora{\ell_1},\ora{\ell_2})$ are R-parallel, that is, $\ora{\ell_2}$ is nested inside $\ora{\ell_1}$ (so we again
have a box of type 2 or 3), as in
Figure~\ref{fig:horizontal-accordion} define the
\emph{horizontal accordion} $F(\vec{p_1},q_1, p_2,q_2)$ by
\[
  p_1^{(k)}\coloneqq p_1 P_2x^k p_2,\qquad
  p_2^{(k)}\coloneqq p_2,\qquad
  q_1^{(k)}\coloneqq q_1,\qquad
  q_2^{(k)} \coloneqq q_2.
\]
\end{definition}


\begin{proposition}
\label{prop:finalfamily}
  Let $F = F(p_1,q_1,p_2,\vec{q_2})$ or $F(\vec{p_1},q_1, p_2,q_2)$ be a
  vertical or horizontal accordion as above. Then
  \begin{itemize}
  \item each $F(k)$ is a RH box;
  \item $F(k+1) \subset F(k)$; and
  \item $\displaystyle \lim_{k \to \infty} \wh\mu_f(F(k)) = 0$.
  \end{itemize}
\end{proposition}

\begin{proof}
  We first prove the results in the case of vertical accordions,
  starting with checking that $F(k)$ is a box. Since $\ora{r_2}$ is
  nested inside $\ora{r_1}$, the NS dynamics of the group element $r_1
  = q_1 x Q_1$ translate $\ora{r_2}$ inside $(r_1^-,r_1^+)$, as depicted
  in Figure~\ref{fig:vertical-accordion}. In particular the cyclic
  order of the endpoints doesn't change, and we also automatically
  have $F(k+1) \subset F(k)$.

  To show the measures go to zero, we first reformulate the nesting
  condition algebraically. Note that
  $q_2Q_1 \cdot \ora{r_1} = \ora{r_2}$. Then  $\ora{r_2}$ and
  $\ora{r_1}$ being nested means $(\ora{r_1},\ora{r_2})$ are
  R-parallel which is equivalent, by Lemma~\ref{lem:rhs}, to
  $(\ora{q_2Q_1}, \ora{r_1})$ R-crossing. This is
  in turn equivalent to $Q_1 \cdot \ora{q_2Q_1} = \ora{Q_1q_2}$
  R-crossing $Q_1 \cdot \ora{r_1} = \ora{x}$; write $e \coloneqq Q_1q_2$.

  In the RH orbit notation, we then have
  \begin{align*}
    a^{(k)} &= p_1 q_1 = a &
    b^{(k)} &= p_1 q_2^{(k)} = p_1 q_1 x^k Q_1q_2 = a x^k e\\
    c^{(k)} &= p_2 q_1 = c &
    d^{(k)} &= p_2 q_2^{(k)} = p_2 q_1 x^k Q_1 q_2 = c x^k e.
  \end{align*}

We thus get the family of RH orbits given by
\[
\bigl([B]_{a,a x^k e,c,c x^k e}\bigr)_{k=0}^\infty.
\]

To show this family of RH orbits have measure going to zero, we
evaluate:
\begin{align*}
  \lim_{k \to \infty} \bigl(\wh\mu_f([B]_{a,ax^ke,c,cx^ke})\bigr)
   &= \frac{1}{4}\lim_{k \to \infty} \lim_{n\to \infty} f([ax^ke x^n]) +
     f([cx^n])-f([ax^n])-f([cx^kex^n]) \\
  &= \frac{1}{4}\lim_{k,n \to \infty} \bigl(f([x^n a]) + f([x^ke])\bigr) + f([cx^n])\\
  &\qquad\qquad\qquad-f([ax^n])-\bigl(f([x^nc]) + f([x^ke])\bigr)\\
  &= 0.
\end{align*}
Here, after applying the definition of $\wh\mu_f$, we apply
Lemma~\ref{lem:additivity-limit-2}
(adding and subtracting suitable copies of $f([x^n])$ and $f([x^k])$)
to see the double limit
exists. Then by applying Proposition~\ref{prop:join-split-alg} twice we
can break up the limit into canceling terms as indicated.

The case of the horizontal accordion is similar, but there are some
confusing inverses so we spell it out briefly.
Since $(\ora{\ell_1},\ora{\ell_2})$ are R-parallel, that is, $\ora{\ell_2}$ is nested inside $\ora{\ell_1}$, the NS dynamics of $\ell_2 = P_2 x p_2$ translate
$\ell_1$ inside $(\ell_1^-,\ell_2^-)$, as in
Figure~\ref{fig:horizontal-accordion}, and the $F(k)$ are boxes with
$F(k+1) \subset F(k)$.
For the limiting computation, since
$P_2 p_1 \cdot \ora{\ell_1} = \ora{\ell_2}$ we have
$(\ora{P_2p_1}, \ora{\ell_2})$ R-cross and thus
$(\ora{p_1 P_2}, \ora{x})$ R-cross.

Set $h \coloneqq p_1 P_2$, so that
\begin{align*}
  a^{(k)} &= p_1^{(k)} q_1 = h x^k c &
  b^{(k)} &= p_1^{(k)} q_2 = h x^k d\\
  c^{(k)} &= p_2 q_1 = c &
  d^{(k)} &= p_2 q_2 = d.
\end{align*}
Lemma~\ref{lem:additivity-limit-2} and
Proposition~\ref{prop:join-split-alg} apply as before.
\end{proof}

is countably additive.
Recall that a RH box~$B$ is by definition half-open, of the form
$[s_1,s_2) \times [t_2,t_1)$. Let
$\overline{B} \coloneqq [s_1,s_2] \times [t_2,t_1]$ and
$B^\circ \coloneqq (s_1,s_2) \times (t_2,t_1)$ be the corresponding
closed and open boxes.

\begin{lemma}\label{lem:extend1}
  For $B = [s_1,s_2)\times [t_2,t_1)$ a RH box and
  any $\epsilon > 0$, there is a RH box $V(B,\epsilon) =
  [s_1',s_2) \times [t_2',t_1)$ so that
  $s_1' < s_1$ and $t_2' < t_2$ and
  $\wh\mu_f(V(B,\epsilon) \setminus B) < \epsilon$.
\end{lemma}
See Figure~\ref{fig:ca1}. That is, $V(B,\epsilon)$ extends $B$ slightly down and to the left,
without increasing the mass much. We write $V^\circ(B,\epsilon)\coloneqq (s_1',s_2)\times(t_2',t_1)$ for the interior of $V(B,\epsilon)$.

\begin{proof}
  Let $B = B(p_1,q_1,p_2,q_2)$.
  By Lemma~\ref{lem:diagonaldense}, there is some $q_3$ so that $q_3
  \cdot \ora{x}$ is nested inside $q_2 \cdot \ora{x}$; in particular
  $F_1 = F(p_1,q_2,p_2,\vec{q_3})$ is a vertical accordion. Furthermore, for
  each $k$, $B \cup F_1(k)$ is a RH box. 
  By Proposition~\ref{prop:finalfamily}, there is $k_1$ large enough
  so that
  $\wh\mu_f(F_1(k_1))<\epsilon/2$.
  Repeat the procedure to find a horizontal
  accordion $F_2$ abutting on $B \cup F_1(k_1)$. Again there is $k_2$
  large enough so that $\wh\mu_f(F_2(k_2)) < \epsilon/2$, and set $V(B,\epsilon) \coloneqq B \cup F_1(k_1) \cup F_2(k_2)$.
\end{proof}

\begin{lemma}\label{lem:extend2}
  For $B = [s_1,s_2) \times [t_2,t_1)$ a RH box and
  any $\epsilon > 0$, there is some $s_2',t_1'$ with
  $s_1 < s_2' < s_2$ and $t_2 < t_1' < t_1$ so that the region
  $W(B,\epsilon) = \bigl([s_1,s_2) \times [t_1',t_1)\bigr) \cup
  \bigl([s_2',s_2) \times [t_2,t_1)\bigr)$ is in $\mathcal{R}$ and satisfies
  $\wh\mu_f(W(B,\epsilon)) < \epsilon$.
\end{lemma}
See Figure~\ref{fig:ca2}. That is, $W(B,\epsilon)$ is a neighborhood of the upper and right
edges of~$B$ with small mass.

\begin{proof}
  Find $q_3$ so that $q_3 \cdot \ora{x}$ is nested inside
  $q_1 \cdot \ora{x}$ and so that $F_1 = F(p_1,q_1,p_2,\vec{q_3})$ is a
  vertical accordion. Similarly find $p_3$ so that $P_2 \cdot \ora{x}$ is nested inside $P_3 \cdot \ora{x}$ and
  $F_2 = F(\vec{p_3},q_1,p_2,q_2)$ is a horizontal accordion. Then
  for sufficiently large $k$, we have $\wh\mu_f(F_1(k)) < \epsilon/2$ and
  $\wh\mu_f(F_2(k)) < \epsilon/2$. Then set
  $W(B,\epsilon) = F_1(k) \cup F_2(k)$.
\end{proof}
  
In the setting of Lemma~\ref{lem:extend2}, also set
\[
W^\circ(B,\varepsilon):=
\bigl([s_1,s_2]\times(t'_1,t_1]\bigr)
\cup
\bigl((s'_2,s_2]\times[t_2,t_1]\bigr);
\]
this is an open neighborhood of the upper and right edges of
$\overline{B}$ in the subspace topology of $\overline{B}$. Note that
neither $W(B,\epsilon)$ nor $W^\circ(B,\epsilon)$ is a subset of the
other.

\begin{proposition}\label{prop:countablyadditive}
  The finitely additive measure $\wh\mu_f$ is countably sub-additive on $\cB$.
\end{proposition}

\begin{proof}
  Let $B_0$ be a right-handed box, and suppose we are given a
  countable covering of it: $B_0 \subset \bigcup_{i=1}^\infty B_i$. We
  must show that
  \[
    \wh\mu_f(B_0) \le \sum_{i=1}^\infty \wh\mu_f(B_i).
  \]
  Fix $\epsilon > 0$. For $i \ge 1$, set
  $V_i \coloneqq V(B_i, \epsilon/2^i)$. Add one more set by setting
  $V_0 \coloneqq W(B_0,\epsilon)$, $V_0^\circ \coloneqq W^\circ(B_0,\epsilon)$ and $V_i^\circ \coloneqq V^\circ(B_i,\varepsilon/2^i)$.

We claim $\{V_i^\circ\}_{i=0}^\infty$ cover $\overline{B_0}$. Indeed,
the sets $V_i^\circ$ for $i\geq1$
cover $B_0$, since
$B_i\subset V_i^\circ$;
in addition
$\overline{B}_0\setminus B_0\subset V_0^\circ$.
Moreover, each $V_i^\circ\cap\overline{B}_0$
is open in the subspace topology of $\overline{B}_0$. Since
$\overline{B}_0$ is compact, there is a finite set
$I\subset\mathbb N\cup\{0\}$ such that
\[
\overline{B}_0\subset\bigcup_{i\in I}V_i^\circ.
\]
For simplicity assume that $0\in I$, adding it in if necessary. It
follows that $B_0\subset \bigcup_{i\in I} V_i$; here we use that
$V_0^\circ \cap B_0 \subset V_0 \cap B_0$.

  By finite additivity, we then have
  \begin{align*}
    \wh\mu_f(B_0) &\le \sum_{i \in I} \wh\mu_f(V_i)\\
      &\le \sum_{i \in I, i \ne 0} \wh\mu_f(B_i) + \sum_{i \in I, i \ne 0}
        \wh\mu_f(V_i \setminus B_i) + \wh\mu_f(V_0)\\
      &< \sum_{i=1}^\infty \wh\mu_f(B_i) + 2\epsilon.
  \end{align*}
  Since $\epsilon$ was arbitrary, we are done.
\end{proof}

We now put everything together to obtain an oriented geodesic current,
following the standard argument that measure on boxes extends to the
Lebesgue measure

\begin{proposition}
Let $f \colon \Curves^+(S) \to \mathbb{R}$ be a functional on curves satisfying symmetry, additivity, smoothing and stability.
Then $\wh\mu_f$ extends to an oriented geodesic current $\overline\mu_f$.
\end{proposition}
\begin{proof}
By Propositions \ref{prop:limitexists}, \ref{prop:RHsemiring},
\ref{prop:finitely-additive}, and~\ref{prop:countablyadditive}, $\wh\mu_f$
is a finitely
additive and countably sub-additive pre-measure on $\mathcal{B}$, so
by Proposition~\ref{prop:addsemitoring},
$\wh \mu_f$ is countably additive on~$\cR$.
 Finally, $\wh\mu_f$ is non-negative by
Proposition~\ref{prop:box-positive}. Since every box has finite $\wh\mu_f$-measure and the countably many
boxes in $\mathcal B$ cover $G(S)$ (Proposition~\ref{prop:density}),
the pre-measure $\wh\mu_f$ is $\sigma$-finite on $\cR$ in the sense of
Theorem~\ref{thm:caratheodory}.
By the Carath\'eodory
Extension Theorem~\ref{thm:caratheodory}, $\wh\mu_f$ has a unique
extension to $\sigma(\cR)$, which we denote~$\overline\mu_f$.

By Proposition~\ref{prop:density}, for every
$\xi\in G(S)$ and every open neighborhood $U$ of $\xi$, there
is an RH box $B\in\mathcal B$ such that
$\xi\in B^\circ\subset B\subset U.$
Since $\pi_1(S)$ is countable, $\mathcal B$ is countable. Consequently,
every open set is a countable union of RH boxes, and hence
$\sigma(\mathcal B)=\operatorname{Bor}(G(S))$, the Borel
$\sigma$-algebra of $G(S)$.
Thus $\overline\mu_f$ is a Borel measure.
Moreover, every point has an open neighborhood contained in an RH box
of finite $\overline\mu_f$-mass. A compact set therefore admits a finite
cover by such neighborhoods, so $\overline\mu_f$ is locally finite. 

To check invariance, note that for every $\gamma\in\pi_1(S)$, the measures
$\gamma_*\overline\mu_f$ and $\overline\mu_f$ agree on $\mathcal B$, by the
$\pi_1(S)$-invariance of $\wh\mu_f$. Since both are
$\sigma$-finite extensions of that premeasure, uniqueness in the
Carath\'eodory extension theorem gives
$\gamma_*\overline\mu_f=\overline\mu_f.$
\end{proof}

It is not \emph{a priori} clear from the definition of
$\overline\mu_f$ that it is a flip-invariant geodesic current
(recall
Definition~\ref{def:currents}). A sketch of a proof is given in Lemma~\ref{lem:RH_flip_invariance}, showing that $\overline \mu_f$ is flip-invariant on RH boxes, from where it follows that $\overline \mu_f$ is flip-invariant.

Rather than relying on that sketch, we instead define the
flip-invariant geodesic current
\begin{equation}
\label{eq:flip-invariant}
\mu_f\coloneqq\OneHalf\bigl(\overline\mu_f+\sigma_*\overline\mu_f\bigr),
\end{equation}
where $\sigma_*$ the push-forward map on oriented geodesic currents
induced from the flip map.

\subsection{Recovering \texorpdfstring{$f$}{f} on curves}
\label{sec:recovery}

It remains to show that $\mu_f$ recovers $f$ as an intersection number on curves.
Before proceeding, we detour to define an asymmetric version of intersection
number between oriented geodesic currents.
\begin{definition}\label{def:asymmetric}
  The \emph{asymmetric intersection} number between oriented geodesic currents is
\begin{equation}
\vec{\imath}(\mu,\nu) \coloneqq \int_{\overline{\mathcal{F}}} \mathbf{1}_R \, d(\mu \otimes \nu),
\label{eq:asymmetric}
\end{equation}
where $\cF$ is a fundamental domain of the action of $\pi_1(S)$ on
$\DG(S)$, as in Definition~\ref{def:intersection_currents}, and
the function $\mathbf{1}_R \colon \DG(S) \to \mathbb{R}$ is
\[
\mathbf{1}_R(\gamma_1,\gamma_2) \coloneqq
\begin{cases}
1 & \text{if } (\gamma_1,\gamma_2)\text{ are R-crossing},\\
0 & \text{otherwise}.
\end{cases}
\]
\end{definition}
From the definitions, we have a relation to the standard intersection
number~\eqref{eq:intersection}:
\begin{equation}
i(\mu, \nu) = \vec{\imath}(\mu, \nu) + \vec{\imath}(\mu, \sigma(\nu)).
\label{eq:relation_intersections}
\end{equation}
For $p,q \in \wt{S}$, partition $G[p,q]$ by
\begin{equation}\label{eq:partition_oriented}
G[p,q) = \ora{G[p,q)} \sqcup \ole{G[p,q)} 
\end{equation}
where  $\ora{G[p,q)}\subset G(S)$ for the \emph{right-oriented} transversals, i.e., the subset of
oriented geodesics in $G[p,q)$ which right-cross the bi-infinite
oriented geodesic~$\ora{l}$ obtained by extending $[p,q) \subset
\wt{S}$. Similarly, we will denote by $\ole{G[p,q)}$ the
\emph{left-oriented} transversals, intersecting the inverted geodesic
$\ora{L}$ to the right. By definition, $\sigma(\ora{G[p,q)})= \ole{G[p,q)}$.

We can compute intersections with curves as follows.

\begin{lemma}
\label{lem:asymmetric_intersection_current}
  For $\vec C = [y]$ an oriented closed curve, choose
$z \in \partial_{\infty} S \setminus \{ y_-, y_+\}$ so that
$y^+,\, y \cdot z,\,z,\,y^-$ are in cyclic order.
The standard intersection and asymmetric intersection
of an oriented closed curve $\vec C=[y]$ with an oriented geodesic current $\mu$ can be computed as 
\begin{align*}
i(\mu, \delta_C) &=  \mu([y \cdot z,z) \times (y^-,y^+)) +
                   \mu(\sigma([y\cdot z,z) \times (y^-,y^+))\\
\vec{\imath}(\mu, \delta_{\vec C}) &= \mu([y \cdot z,z) \times (y^-,y^+)).
\end{align*}
\end{lemma}
Recall that $\delta_C$ is the geodesic current associated to the unoriented curve (or flip-invariant)~$C$, while $\delta_{\vec C}$ denotes the geodesic current representing the oriented curve $\vec C$.

Note that
$[y\cdot z,z)\times (y^-,y^+)$ is a Borel fundamental domain for the
  action of $\langle Y \rangle$ on the set of geodesics that intersect
  $\ora{y}$ transversely.
\begin{proof}
Choose $z\in(y^+,y^-)$ so that $y^+,\,y\cdot z,\,z,\,y^-$
occur in cyclic order, and set
\[
A_y\coloneqq [y\cdot z,z)\times(y^-,y^+).
\]
Then $A_y$ is a Borel fundamental domain for the action of
$\langle y\rangle$ on the oriented geodesics which cross
$\ora y$ transversely (See~\cite[Lemma~4.8]{HP97:RigidityNegCurvedCone}): the proof only uses
the $\pi_1(S)$-invariance of $\mu$.) Hence
$\vec i(\mu,[y])=\mu(A_y)$,
and, using the opposite crossing orientation,
$i(\mu,[y])=\mu(A_y)+\mu(\sigma A_y)$. The same claim follows for the intersection number using Equation~\ref{eq:relation_intersections}.
\end{proof} 

Analogously, an asymmetric intersection number between curves can be defined as follows.

\begin{definition}\label{def:asym-intersect}
  Given two oriented curves $\vec C$, $\vec D$, the
  asymmetric intersection number $\vec{\imath}(\vec C, \vec D)$ is the
  minimal number of \emph{right-handed} crossings between any two homotopy
  class representatives of $\vec C$ and~$\vec D$, i.e., the number of
  points on~$S$ where $\vec C$ crosses $\vec D$ to the right in the
  sense of Definition~\ref{def:crossing}.
\end{definition}

It is straightforward to see that the asymmetric intersection number
  from Definition~\ref{def:asymmetric} satisfies
  $\vec{\imath} (\delta_{\vec C}, \delta_{\vec D}) = \vec\imath(\vec C, \vec D)$ and, for a flip-invariant curve $C$ and any oriented geodesic current $\mu$, we have
  \[
  \vec{\imath}(\mu, \delta_{C}) = \OneHalf i(\mu, \delta_C).
  \]
In particular,
\[
\vec{\imath}(\delta_C, \delta_D) = \OneHalf i(C,D)
\]
We finish by proving Theorem~\ref{thm:intersect}, first in the following
form.

\begin{theorem}
  \label{thm:recoverintersection}
Let $f \colon \Curves^+(S) \to \mathbb{R}$ be a curve functional
satisfying symmetry, smoothing, additivity and stability, and let
$\mu_f$ be as above.
For $\vec C$ an oriented closed curve, we have
\[
i(\mu_f, \delta_C) =
  2\vec\imath(\mu_f, \delta_{\vec C}) = f(\vec C).
\]
\end{theorem}
\begin{proof}
  We first prove that
  $\vec{\imath}(\mu_f,\delta_{\vec C})=\OneHalf f(\vec C)$.
  Given a representative $y \in \pi_1(S)$ of the conjugacy
  class~$\vec C$, we will find a sequence of
RH boxes to approximating the fundamental domain for $\langle
y\rangle$ above, and then show that computing the measure of that
fundamental domain yields $f([y])$.

Choose $p,q \in \pi_1(S)$ so that $(P\cdot \ora{x}, \ora{y},
q\cdot\ora{x})$ are R-parallel, as in Figure~\ref{fig:recover-f}.
The NS dynamics of the action of $y$ imply that, for $m_1
> m_2$ and $n_1 > n_2$, the box
$B(py^{-m_1},py^{-m_2},y^{n_1}q,y^{n_2}q)$ form a RH box. We will look
at the following particular family for $n \ge 0$:
\begin{align*}
  p_1^{(n)} &\coloneqq p & q_1^{(n)}&\coloneqq y^n q\\
  p_2^{(n)} &\coloneqq p y &q_2^{(n)} &\coloneqq y^{-(n+1)} q.
\end{align*}

\begin{figure}
\centering{
\svgfignormal{recover-f}}
\caption{The boxes used in proving that $\mu_f$ recovers $f$.}
  \label{fig:recover-f}
\end{figure}

In the RH orbit notation, we have
\begin{align*}
  a^{(n)} &= p y^{n}   q &  b^{(n)} &= p y^{-(n+1)} q \\
  c^{(n)} &= p y^{n+1} q &  d^{(n)} &= p y^{-n} q.
\end{align*}

If we let $z=(P \cdot \ora{x})^-$, and $B_n \coloneqq B_{a^{(n)},
  b^{(n)}, c^{(n)}, d^{(n)}}$ we then have
    \begin{align*}
      \vec{\imath}(\overline\mu_f, \delta_{[y]})&=\overline\mu_f([z,Yz)\times(y^-,y^+)) \\
    &= \lim_{n\to\infty} \wh\mu_f(B_n)  \\
    &= \frac{1}{4}\lim_{n\to\infty}\lim_{m\to\infty}f([py^{n+1}qx^{m}]) + f([py^{-(n+1)}qx^{m}])-f([py^{n}qx^{m}]) - f([py^{-n}qx^{m}]) \\
    &= \OneFourth\bigl(f([y]) + f([y^{-1}])\bigr) =\OneHalf f([y]).
    \end{align*}
    The first equality follows by the fact that the boxes defined
    above converge from below to the  $[z,Y
      \cdot z) \times (y^-,y^+)$ (the fundamental domain for the
  action of $\langle Y \rangle$) as $n \to \infty$, since $B_{n-1} \subset B_n$.
    At the second equality, we use definition of $\wh\mu_f$ on RH boxes.
    For the third, Lemma~\ref{lem:additivity-limit-2} lets us pass
    from the iterated limit to a double limit, and gives a
    number of canceling terms of $mf([x])$ and $nf([y^{\pm 1}])$; only
    two terms survive the cancellation. Then symmetry of $f$ gives the result.

We then get $\vec\imath(\sigma_*\overline\mu_f,\delta_{\vec C})
=\vec\imath(\overline\mu_f,\delta_{\vec C^{-1}})=\frac12 f(\vec C)$, using the definition of $\sigma_*$, and the symmetry of $f$.
Since $\mu_f$ is flip-invariant, Lemma~\ref{lem:asymmetric_intersection_current}
gives
$i(\mu_f,\delta_C)=f(\vec C)$, as desired.
\end{proof}

\begin{remark}
In \cite[Proposition~4.3]{BIPP24:PositiveCrossratios} a geodesic current is induced
from a cross-ratio.
Our construction is more general (in particular, we do not rely on a cross-ratio).
We discuss here why our construction does not follow from their
arguments once we have defined $\mu_f$ on boxes.
Given a dense subset $X \subset \partial_{\infty} S$, they consider the set of $\mathcal{R}(X)$ of all proper rectangles of geodesics with endpoints on $X$.
They start from a finitely additive function $r \colon \mathcal{R}(X) \to \mathbb{R}$ (defined suitably from a cross-ratio)
that is (1) side-by-side additive: if $R,R_1,R_2 \in \mathcal{R}(X)$
with $R=R_1 \cup R_2$, with $R_1,R_2$ disjoint, $r(R)=r(R_1)+r(R_2)$;
and (2) monotonic: if $R,R' \in \mathcal{R}(X)$ with $R \subset R'$, we have $r(R) \leq r(R')$.

Our function $\mu_f$ on RH boxes satisfies these properties by Proposition~\ref{prop:sidebyside}
and Proposition~\ref{prop:finitely-additive}, however, if we take $X$ to
be the set of
endpoints of translates of~$x$, then $\mu_f$ is only
defined on a proper subset of $\mathcal{R}(X)$.
Therefore, we cannot directly apply their result.

Moreover, we need the extra control on the family of RH
boxes to recover the curve functional~$f$ from the geodesic current.
In Section~\ref{sec:crossratios}, we prove that every cross-ratio
induces a curve functional (i.e., period) satisfying the hypotheses of
our Theorem~\ref{thm:intersect}, and thus our result reproves
\cite[Proposition~4.3]{BIPP24:PositiveCrossratios} for closed surfaces.
\end{remark}



\subsection{Proof of  Main Theorems~\ref{thm:intersect} and~\texorpdfstring{\ref{thm:intersections_group}}{A'}}

Now we prove the two versions of the main theorem as stated in the introduction. The topological version (Theorem~\ref{thm:intersect}) is stated in terms of functionals on unoriented multi-curves. The algebraic version of the
theorem, namely, Theorem~\ref{thm:intersections_group}, is phrased in terms of functions on the surface group.

\begin{proof}[Proof of Theorem~\ref{thm:intersect}]
  Given a functional~$f$ as in the statement, by
  Definition~\ref{def:smoothing} and
  Proposition~\ref{prop:power_disconnected_imply_stable}, the induced
  additive and symmetric functional on oriented multi-curves satisfies
  stability. Hence the existence of $\mu_f$ follows by
  Theorem~\ref{thm:recoverintersection}.


  Uniqueness of the geodesic current $\mu_f$ is a result of Otal,
  geodesic currents are determined by their marked length spectrum
  \cite[Th\'eor\`eme~2]{Otal90:SpectreMarqueNegative}.
  The claim about the topology is due to Duchin-Leininger-Rafi, who
  show that the map $\Curr(S) \to \bbR^{\Curves(S)}$
  defined by $\mu \mapsto i(\mu, \cdot)$ is a homeomorphism onto its
  image \cite[Theorem~12]{DLR10:DegenerationFlatMetrics}.
\end{proof}

\begin{remark}
  Uniqueness is not true if we don't assume $\mu_f$ is flip-invariant.
  See Appendix~\ref{sec:asymmetric}.
\end{remark}

\begin{proof}[Proof of Theorem~\ref{thm:intersections_group}]
As in the statement, let $\overline{f}(a) \coloneqq
\frac{1}{2}\left(f(a) + f(A) \right)$, and extend it linearly to be an
additive
function on multi-curves. Since $\overline{f}$ is
symmetric, its oriented smoothing inequalities imply the corresponding
unoriented smoothing inequalities by Proposition~\ref{prop:oriented_smoothing_unoriented}. Stability gives
power smoothing:
\[
\overline{f}(a^{n+m})=\overline{f}(a^n)+\overline{f}(a^m).
\]
Thus $\overline{f}$ satisfies the hypotheses of Theorem~\ref{thm:intersect}, which produces the
required geodesic current.

Conversely, if $\overline{f}(g)=i(\mu,[g])$, then class invariance, stability,
and the smoothing inequalities follow from the corresponding
properties of intersection with a geodesic current; see
Proposition~\ref{prop:necessary}.

To prove the sufficient criterion, we will show that, assuming the oriented smoothing hypotheses on $f$, the extension of $\overline{f}$ satisfies the hypothesis of Theorem~\ref{thm:intersect}.

We will first verify smoothing of $\overline{f}$.
We start
with oriented disconnected smoothing, between crossing elements
$(\ora{a},\ora{b})$, which we can assume R-cross by switching $a,b$ if
necessary. We have
\begin{align*}
  \overline{f}(a) + \overline{f}(b)
    &= \frac{1}{2}\bigl(f(a) + f(A) + f(b) + f(B)\bigr)\\[2pt]
    &\ge \frac{1}{2}\bigl(f(ab) + f(AB)\bigr) = \overline{f}(ab).
\end{align*}
In the inequality, we used the assumption on $f$ and the fact that
$(\ora{A},\ora{B})$ also
R-cross, as well as conjugation invariance for $f(AB) = f((ba)^{-1})$.
By grouping the four terms differently, and using the fact that
$(\ora{b},\ora{A})$ R-cross, we also get $\overline{f}(a) +
\overline{f}(b) \ge \overline{f}(bA)$.

Next we turn to oriented connected smoothing of a curve at an
essential self-crossing.
Using Proposition~\ref{prop:crossings-triality}, assume the essential
self-crossing is realized by the self-intersection of a curve
$[\gamma]=[ab]$, where $(\ora{a},\ora{b})$ are parallel; by switching
them if necessary, we can assume they are R-parallel, and thus so are
$(\ora{B},\ora{A})$. Then
\begin{align*}
  \overline{f}(ab)
    &= \frac{1}{2}\bigl(f(ab) + f(BA)\bigr) \\[2pt]
    &\ge \frac12\bigl(f(a) + f(b) + f(B) + f(A)\bigr)
     = \overline{f}(a) + \overline{f}(b).
\end{align*}

We have thus deduced $\overline{f}$ satisfies oriented connected and disconnected smoothing. Therefore, by Proposition~\ref{prop:oriented_smoothing_unoriented} below,
$\overline{f}$~satisfies unoriented connected and disconnected smoothing as well. 
Finally, stability implies oriented power-smoothing, i.e., $f(a^{n+m}) \geq f(a^n) + f(a^m)$ for every $n, m \geq 1$. This establishes smoothing of $\overline{f}$. Hence, smoothing, stability and additivity are satisfied for the function on unoriented curves $\overline{f}$.
We are thus under the
hypotheses of Theorem~\ref{thm:intersect}.
\end{proof}

\begin{remark}
In fact,  we can replace the following hypotheses in~Theorem~\ref{thm:intersections_group} by the following slightly weaker ones:
\begin{itemize}
 \item Stability can be replaced by power smoothing (by Proposition~\ref{prop:power_disconnected_imply_stable});
\item oriented smoothing can be replaced by R-oriented smoothing:
 \begin{align}
f(ab) &\ge f(a) + f(b)
&& \text{if } (\ora{a},\ora{b}) \text{ are R-parallel}, \\
f(a) + f(b) &\ge f(ab)
&& \text{if } (\ora{a},\ora{b}) \text{ R-cross}.
\end{align}
\end{itemize}
\end{remark}
