\section{Construction of the geodesic current}
\label{sec:construct}



\subsection{\texorpdfstring{$\protect{\mu_{f}}$}{muf} is countably additive}
\label{sec:countable-additivity}

We prove countable additivity of the function $\mu_{f}$ by
proving approximation results.



\begin{definition}
Let the right handed lift system $[p_1,q_1,p_2,q_2]$ determine a right-handed box $B = [a_1,a_2) \times [b_1,b_2)$.
We define a \emph{top vertical accordion} of boxes based at $B$ the sequence of boxes, indexed by $k$, given for each $k$, by
\[
p_1^{(k)}=p_1,p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)} \coloneqq q_1 x^k q_1^{-1}q_2.
\]
Since the endpoint corresponding to $q_2$ is the one changing, we will denote these sequence of boxes by
\[
F_{p_1,p_2,q_1,\vec{q_2}}.
\]
This defines boxes whose vertical direction becomes narrower as $k$ grows, with the top endpoint fixed. 
Similarly, we define \emph{bottom vertical accordion}, and denote it by 
\[
F_{p_1,p_2,\bar{q_1},q_2},
\]
giving a sequence of boxes whose vertical direction becomes narrower and the bottom endpoint fixed, i.e.,
\[
p_1^{(k)}=p_1,p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)} \coloneqq q_1 x^k q_1^{-1}q_2.
\]

On the other hand, if we fix $p_2,q_1,q_2$ and allow $p_1$ to vary, by choosing nested lifts of $x$ for $p_1$ and $p_2$, we obtain a datum given by the box notation $p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)}=q_2$, and
\[
p_1^{(k)} = p_2 x^k p_2^{-1} p_1.
\]
and we denote the corresponding family of boxes,
\[
F_{\vec{p_1},p_2,q_1,q_2}.
\]
We will call them a \emph{right horizontal accordion of boxes based at $B$} (the right endpoint is fixed).
Similarly, we can define a \emph{left horizontal accordion} (the left endpoint is fixed).
\label{def:accordion}
\end{definition}

The following Proposition will be key to control how much $\mu_f$ when the right-handed boxes are varied in a control way. In particular, it will be crucial in proving countable addivitity of $\mu_f$.

\begin{proposition}
 Given a RH system $[p_1,p_2,q_1,q_2]$ so that, moreover, $q_1^{-1}q_2$ cross $x$ to the right. let $F(k)$ be  a system of lifts in an accordion configuration, as in Definition~\ref{def:accordion}, in any of the four types $F_{\vec{p_1},p_2,q_1,q_2}(k)$, $F_{p_1,p_2,q_1,\vec{q_2}}$, $F_{p_1,\vec{p_2},q_1,q_2}(k)$ and $F_{p_1,p_2,\vec{q_1},q_2}(k)$. We have

\begin{itemize}
 \item For every $k \geq 0$, $F(k)$ is a right-handed box.
\item For every $k$, $F(k+1) \subset F(k)$
\item
\[
\lim_{k \to \infty} \mu_f(F(k))= 0
\]
\end{itemize}
\label{prop:finalfamily1}
\end{proposition}
\begin{proof}
\begin{itemize}
    \item We first prove that an accordion $F(k)$ is a right-handed box. Consider first a top vertical accordion $F_{\vec{p_1},p_2,q_1,q_2}(k)$ (the other types are similar).
    We know that $[p_1,p_2,q_1,q_2]$ form a RH system, so 
    $p_i q_i$ cross $x$ to the right
        and $q_2 x^n p_1$ cross $q_1x^n p_2$ to the right (by Definition~\ref{def:crossing}). 
        By north-south dynamics of $q_1 x q_1^{-1}$, the translates of $\tilde{x}$ look like in Figure~\ref{fig:finalfamily}.
        The following properties follow immediately.
    
    \begin{enumerate}
        \item $p_1 q_1 x^k q_1^{-1} q_2$ crosses $x$ to the right.
        \item $p_2 q_1 x^k q_1^{-1} q_2$ crosses $x$ to the right.   
        \item   $q_1 x^k q_1^{-1} q_2x^n p_1$ cross $q_1x^n p_2$ to the right.
    \end{enumerate} 
    This proves that top vertical accordions are right handed boxes. The proof is similar for other types of accordions.
    \item We prove that for a fixed RH sytem $[p_1,p_2,q_1,q_2]$ and a corresponding accordion $F(k)$, $\lim_{k \to \infty} \mu(F(k))=0$.
    We give the argument for type $F_{p_1,p_2,q_1,\vec{q_2}}$, the other types are similar.
    We will first go from the box notation to the orbit notation, since the latter is better suited for the definition of $\mu_f$.
    We have
 \begin{align*}
     a &= p_1 q_1 \\
     b &= p_1 q_1 x^k q_1^{-1} q_2 \\     
     c &= p_2 q_1  \\  
     d &= p_2 q_1 x^k q_1^{-1} q_2 \\  
 \end{align*}
 Writing $e=q_1^{-1} q_2$, we have
 \begin{align*}
     b &= a x^k e \\     
     d &= c x^k e. \\  
 \end{align*}
We thus get the family of box orbits given by
\[
\bigl([B]_{a,a x^k e,c,c x^k e}\bigr)_{k=0}^\infty.
\]
Figure~\ref{fig:finalfamily} related the box to the orbit notation.

    We want to show
\[
\lim_{k \to \infty} \mu_f([B]_{a,a x^k e,c,c x^k e})=0.
\]
By definition of $\mu_f$,
\begin{align*}
\mu_f([B]_{a,a x^k e,c,c x^k e})=\lim_{n\to \infty} f([ax^ke x^n]) + f([cx^n])-f([ax^n])-f([cx^kex^n]).
\end{align*}
So, we are left to evaluate the nested limit
\begin{align*}
\lim_{k \to \infty} \lim_{n\to \infty} f([ax^ke x^n]) + f([cx^n])-f([ax^n])-f([cx^kex^n]).
\end{align*}
    We use Proposition~\ref{prop:join-split-geom},
    equations~\ref{eq:cross-join} and
    \ref{eq:cross-split}.
    
    Let $s>0$ be the constant (depending on $a,e$) furnished by Equation~\ref{eq:cross-join}, and let 
    $s'$ be the constant (depending on $c,e$) from Equation~\ref{eq:cross-split}.
    We assume $n$ is large enough so that $n-k-s>s$ and  $n-k-s'>s'$.
    We can then take $r>0$ satisfying the hypotheses of both equations, namely, $n-k-s>r>s$ and $n-k-s'>r>s'$.
    We then have the expression in the above iterated limit is upper bounded by
    \[
      f([ax^{k-r}]) + f([e x^{n+r}]) + f([cx^n])-f([ax^n])-f([cx^{k+r}]) - f([ex^{n-r}])
    \]
    by adding and subtracting a term $f([x^{k+n}])$, the iterated limit of the last expression is equivalent to the iterated limit
    \begin{align*}
       \lim_{k\to\infty} \lim_{n\to\infty}  \bigl( f([ax^{k+1}]) - f([x^{k+1}]\bigr) + \bigl( f([ex^{n-1}]) - f([x^{n-1}])\bigr) +  \\
        \bigl( f([cx^n]) - f([x^n]) \bigr) - \bigl(f([cx^n]) - f([x^n])\bigr)\\
        -\bigl(f([cx^{k+2}])-f([x^{k+2}])\bigr)-\bigl(f([ex^{n-2}])-f([x^{n-2}])\bigr).
    \end{align*}
    Finally, this iterated limit is 0 by  Lemma~\ref{lem:additivity-limit}, since $e=q_1^{-1} q_2$ crosses $x$ to the right, by assumption.\qedhere
    \end{itemize}
\end{proof}

We now use Proposition~\ref{prop:finalfamily1} to prove that $\mu_f$
is countably additive.

\begin{remark}
We remark that at this point of the argument, we could also use a more indirect strategy, using Riesz representation, as done in \cite[Proposition~3.11]{BIPP21:PositiveCrossratios}.
Indeed, the endpoints of r.h. boxes correspond to endpoints of lifts of $x$, and these are dense in $\partial_{\infty} S$ by Lemma~\ref{lem:minimal}. Moreover, if $R,R_1,R_2$ are r.h. boxes with $R=R_1 \cup R_2$, with $R_1,R_2$ disjoint, $\mu_f(R)=\mu_f(R_1)+\mu_f(R_2)$, by Proposition~\ref{prop:sidebyside}.
Also, by the argument in Proposition~\ref{prop:RHsemiring}, for $R,R'$ r.h. boxes, if $R \subset R'$, we have $\mu_f(R) \leq \mu_f(R')$.
Hence, our function $\mu_f$ satisfies the hypotheses of~\cite[Proposition~4.1]{BIPP21:PositiveCrossratios}, and hence by ~\cite[Proposition~4.3]{BIPP21:PositiveCrossratios} we obtain a geodesic current with desired properties.
We choose to give an independent argument that showcases the fact that r.h. boxes enjoy the extra structure of a semi-ring.
\end{remark}
We will need to consider more types of boxes
than the half-open boxes we have been working with so far. For $B$ a right-handed box $[a_1,a_2) \times [b_1,b_2)$, let
$\overline{B} \coloneqq [a_1,a_2] \times [b_1,b_2]$ and
$B^\circ \coloneqq (a_1,a_2) \times (b_1,b_2)$ be the corresponding
closed and open boxes.

\begin{lemma}\label{lem:extend1}
  For $B = [a_1,a_2) \times [b_1,b_2)$ a right-handed box and
  any $\epsilon > 0$, there is a right-handed box $V(B,\epsilon) =
  [a_1',a_2) \times [b_1',b_2)$ in $\mathcal{R}$ so that
  $a_1' < a_1$ and $b_1' < b_1$ and
  $\mu_f(V(B,\epsilon) \setminus B) < \epsilon$.
\end{lemma}

That is, $V(B,\epsilon)$ extends $B$ slightly down and to the left,
without increasing the mass much.

\begin{proof}
  Let $B = [a_1,a_2) \times [b_1,b_2)$. By Proposition~\ref{prop:density} there is a right handed box $B'=[a_1'',a_1) \times [b_1'',b_1)$ in $\mathcal{R}$ so that
  $a_1'' < a_1$ and $b_1'' < b_1$.
  Let $H$ be a right horizontal accordion based at $B'$. 
  By Proposition~\ref{prop:finalfamily1}, for $k$ large enough, we have $\mu_f(F(k))<\epsilon$. 
  Let $V'(B,\epsilon) \coloneqq B \cup F(k)$.
  We can now repeat the same procedure with $V'(B,\epsilon)$ in the vertical direction.
\end{proof}

\begin{lemma}\label{lem:extend2}
  For $B = [a_1,a_2) \times [b_1,b_2)$ a right-handed box and
  any $\epsilon > 0$, there is a region $W(B,\epsilon) =
  \bigl([a_1,a_2) \times [b_1',b_2)\bigr) \cup
  \bigl([a_2',a_2) \times [b_1,b_2)\bigr)$ in $\mathcal{R}$ so that
  $a_1 < a_2' < a_2$ and $b_1 < b_2' < b_2$ and
  $\mu_f(W(B,\epsilon)) < \epsilon$.
\end{lemma}

That is, $W(B,\epsilon)$ is a neighborhood of the upper and right
edges of~$B$ with small mass.

\begin{proof}
Let $H(k)$ be a right horizontal accordion, and $F(k)$ a top vertical accordion.
We tan take $k$ large enough so that $\mu_f(H(k)) < \epsilon/2$,  $\mu_f(F(k)) < \epsilon/2$, and then let $W(B,\epsilon) = B \cup H(k) \cup V(k)$.
\end{proof}
  
In the setting of Lemma~\ref{lem:extend2}, also set
\[
  W^\circ(B,\epsilon) \coloneqq \bigl([a_1,a_2] \times (b_2',b_2]\bigr) \cup
    \bigl((a_2',a_2] \times [b_1,b_2]\bigr);
\]
this is an open neighborhood of the upper-right edges of the
$\overline{B}$ in the subspace topology.

\begin{proposition}\label{prop:countablyadditive}
  The finitely additive measure $\mu_f$ is countably subadditive.
\end{proposition}

\begin{proof}
  Let $B_0$ be a right-handed box, and suppose we are given a
  countable covering of it: $B_0 \subset \bigcup_{i=1}^\infty B_i$. We
  must show that
  \[
    \mu_f(B_0) \le \sum_{i=1}^\infty \mu_f(B_i).
  \]
  Fix $\epsilon > 0$. For $i \ge 1$, set
  $V_i \coloneqq V(B_i, \epsilon/2^i)$. Also add one more set by setting
  $V_0 \coloneqq W(B_0,\epsilon)$ and $V_0^\circ \coloneqq W^\circ(B_0,\epsilon)$.

  Then $\{V_i^\circ\}_{i=0}^\infty$ cover $\overline{B_0}$, since
  the $B_i$ cover $B_0$, $B_i \subset V_i^\circ$, and
  $\overline{B_0}\setminus B_0 \subset V_0^\circ$. Since
  $\overline{B_0}$ is compact and the $V_i^\circ$ are open (in the
  subspace topology), we have a finite subcover; let $I \subset
  \mathbb{N}$ be the set of
  indices included in the subcover.

  By finite additivity, we then have
  \begin{align*}
    \mu_f(B_0) &\le \sum_{i \in I} \mu_f(V_i)\\
      &\le \sum_{i \in I, i \ne 0} \mu_f(B_i) + \sum_{i \in I, i \ne 0}
        \mu_f(V_i \setminus B_i) + \mu_f(V_0)\\
      &< \sum_{i=1}^\infty \mu_f(B_i) + 2\epsilon.
  \end{align*}
  Since $\epsilon$ was arbitrary, we are done.
\end{proof}

We now put everything together to obtain a geodesic current.

\begin{proposition}
Let $f \colon \mathcal{C}(S) \to \mathbb{R}$ be a functional on curves satisfying additivity, smoothing, stability and homogeneity.
Then $\mu_f$ is an oriented geodesic current.
\end{proposition}
\begin{proof}
By Proposition~\ref{prop:limitexists},
Proposition~\ref{prop:RHsemiring}, and
Proposition~\ref{prop:countablyadditive}, $\mu_f$ is a finitely
additive and countably subbadditive pre-measure on $\mathcal{R}$. By covering by an exhaustion consisting of a union of nested boxes, we can see $\mu_f$ is also $\sigma$-finite.
Therefore, by Carathéodory extension Theorem~\ref{thm:caratheodory}, $\mu_f$ can be extend to a measure $\mu_f$ on $\mathcal{A}_{\mathcal{R}}$, the $\sigma$-algebra generated by the semi-ring $\mathcal{R}$.
Let $\mathcal{B}$ be the Borel $\sigma$-algebra of $G(S)$.
Since $\mathcal{A} \subset \mathcal{B}$, we have $\mathcal{A}_{\mathcal{R}} \subset \mathcal{B}$.
On the other hand, given a half-open box of geodesics $B=[a,b) \times [c,d) \in \mathcal{B}$, it can be written as an intersection of two countable unions of boxes in $\mathcal{R}$, since, by Proposition~\ref{prop:density}, the set of elements in $\mathcal{R}$ are dense in the subset of half-open rectangles of $G(S)$ (this is the same as the standard argument one uses to prove the Borel $\sigma$-algebra generated by half-rectangles with rational endpoints is the usual Borel $\sigma$-algebra).
Note that $\mu_f$ is locally finite, since any compact subset of $G(S)$ can be covered by one of the boxes in $\mathcal{R}$, where the measure takes a finite value.
Therefore, $\mu_f$ is a Borel and locally finite measure. 
By definition of the pre-measure $\mu_f$, this measure $\mu_f$ is $\pi_1(S)$-invariant. Finally, $\mu_f$ is non-negative by Proposition~\ref{prop:box-positive}.
\qedhere
\end{proof}

\begin{remark}
It is not a priori clear from the definition of $\mu_f$ that it is an unoriented geodesic current (recall Definition~\ref{def:currents}).
In order to obtain an unoriented geodesic current, we can take
\[
\overline{\mu_f} \coloneqq 1/2(\mu_f + \sigma_*(\mu_f)).
\]
\end{remark}

\subsection{
\texorpdfstring{$\protect\mu_{f}$}{muf} recovers \texorpdfstring{$\protect{f}$}{f} on curves}
\label{sec:recovery}

It remains to show that $\mu_f$ recovers $f$ as an intersection number on curves.

If $y$ is a hyperbolic element representing a closed curve
$C$ in the surface $S$, let
$z \in \partial_{\infty} S - \{ y_-, y_+\}$ so that
$y_-, z, y \cdot z, y_+$ are in cyclic order.
Then
$(y_+,y_-) \times [z,y \cdot z)$ is a Borel fundamental domain for the
$\langle y \rangle$-action on $\DG(S)$ (the set of pairs of crossing geodesics).

First we note that the following.

\begin{lemma}The intersection
of an oriented (or unoriented) closed curve $C$ with an oriented (or unoriented) geodesic current $\mu$ can be computed as
\[
i(\mu, C) = \mu((y_+,y_-) \times [z,y \cdot z)).
\]
\end{lemma}
\begin{proof}
The claim follows from the fact that if $y \in \pi_1(\Sigma)$ denotes the element representing the oriented curve $C$, then for any $q \in \ora{y}$,
\[
\mu[q, y \cdot q) = \mu((y_+,y_-) \times [z,y \cdot z))
\]
for every $z \in \partial_{\infty} S - \{ y_-, y_+\}$ so that
$y_-, z, y \cdot z, y_+$ are in cyclic order.
This is
spelled out in detail in the proof of~\cite[Lemma~4.4]{MZ19:PositivelyRatioed}, (cf. \cite[Page~153]{Otal90:SpectreMarqueNegative}), and the proof only uses the $\Gamma$-invariance of $\mu$, and no flip-invariance.
\end{proof}

We finish by proving Theorem~\ref{thm:intersect}], whose statement we repeat here for convenience.

\begin{theorem}
Let $f \colon \Curves(S) \to \mathbb{R}$ be a curve functional satisfying smoothing, additivity, stability and homogeneity.
For $C$ a closed curve, we have
\[
i(\mu_f,C)=i(\overline{\mu_f},C)=f(C).
\]
In particular, $\overline{\mu_f}$ is unique.
\label{prop:recoverintersection}
\end{theorem}
\begin{proof}
We first prove that $i(\mu_f,C)=f(C)$. Given a representative $y \in \pi_1(S,*)$ of the conjugacy class~$C$, we will first show that we can use right handed boxes to approximate a fundamental domain for $\supp \mu \times \{ \tilde{y} \}$ in $\DG(\Sigma)$.
Second, we'll show that computing the measure of that fundamental domain yields $f([y])$.


Choose a conjugate $y$ in $[y]$, 
and $p_1, q_1 \in \pi_1(S,*)$, so that $x$ crosses $p_1$ to the right, i.e., $p_1^{-1}$ crosses $x$ to the right, and $p_1$ crosses $y$ to the right.
Furthermore, choose $q_1$ so that $q_1$ crosses $x$ and $y$ to the right. 
Then, by Lemma~\ref{lem:rhs}, we have a configuration of lifts like the upper left and right lifts of $\wt{x}$ in Figure~\ref{fig:fundamental}.
Moreover, this choice guarantees that $q_1 p_1$ crosses $y$ to the right and thus, by Lemma~\ref{lem:rhs}, $\wt{y}$ is parallel to $q_1 p_1 \cdot \wt{y}$.

For each $n$, define the following elements of $\pi_1$:
\begin{align*}
  p_1^n &\coloneqq p_1 & q_1^n &\coloneqq y^n q_1\\
  p_2^n &\coloneqq p_1 y &q_2^n &\coloneqq y^{-(n+1)} q_1.
\end{align*}
\begin{claim}
The datum $[p_1^n,p_2^n,q_1^n,q_2^n]$, as defined above, is a RH system, i.e.,
\begin{enumerate}
    \item $p_i^nq_j^n$ cross $x$ to the right, for $i,j=1,2$.
    \item $q_2^n x^m p_1^n$ crosses $q_1^n x^m p_2^n$ to the right, for $m \geq 0$. 
\end{enumerate}
\end{claim}
\begin{proof}
By NS dynamics of $y$, the translates of $\wt{x}$ by $(p_i^{(n)})^{-1}$ and $q_i^{(n)}$, for $i=1,2$, are like in Figure~\ref{fig:fundamental}.
\end{proof}

If we translate the RH system
$[p_1^{(n)},p_2^{(n)},q_1^{(n)},q_2^{(n)}]$ into the orbit notation,
and set $p \coloneqq p_1$ and $q \coloneqq q_1$, we have right-handed boxes
\begin{align*}
  a^{(n)} &= p y^{n}   q &  b^{(n)} &= p y^{-(n+1)} q \\
  c^{(n)} &= p y^{n+1} q &  d^{(n)} &= p y^{-n} q.
\end{align*}

If we let $z=(p_1^{-1} \cdot \wt{x})^-$, and $B_n \coloneqq B_{a^{(n)}, b^{(n)}, c^{(n)}, d^{(n)}}$ then we have
    \begin{multline*}
      \mu_f((\gamma_-,\gamma_+) \times [z,y \cdot z)) \\
      \begin{aligned}
    &= \lim_{n\to\infty} \mu_f(B_n)  \\
    &= \frac{1}{2}\lim_{n\to\infty} \lim_{m\to\infty}f([py^{n+1}qx^{m}]) + f([py^{-(n+1)}qx^{m}])-f([py^{n}qx^{m}]) - f([py^{-n}qx^{m}]) \\
    &=\frac{1}{2}\lim_{n,m\to\infty}(f([py^{n+1}qx^{m}]) - f([y^{n+1}]) - f([x^m])) \\
    &\qquad+ (f([py^{-(n+1)}qx^{m}])-f([y^{-(n+1)}]) -f([x^m])) \\
    &\qquad-(f([py^{n}qx^{m}])-f([y^n])-f([x^m]))\\
    &\qquad- (f([py^{-n}qx^{m}])-f([y^{-n}])-f([x^m])) \\
    &\qquad+ f([y^{n+1}])+f([y^{-(n+1)}]) - f([y^n]) - f([y^{-n}])\\
    &= \frac{1}{2}(f([y]) + f([y^{-1}]))\\
    &=f([y]).
      \end{aligned}
    \end{multline*}      The first equality follows by the fact that the boxes defined
      above converge from below to $(\gamma_-,\gamma_+) \times [z,y
      \cdot z)$ as $n \to \infty$, since $B_{n-1} \subset B_n$. The second equality is by
      definition of the measure $\mu_f$ on boxes.
      For the third, we pass from double limit to each one of the iterated limits by Lemma~\ref{lem:additivity-limit-2} and we add and subtract the necessary powers of~$y$,
      $y^{-1}$ and $x$. The hypotheses of that Lemma are satisfied by the choice of $p_1$ and $q_1$.
      In the fourth equality, we then apply that
      Lemma to cancel out equivalent
      terms. In the fifth equality, we use the symmetry property of~$f$ to obtain the result.

      It remains then to check the property $i(\overline{\mu}_f, C)=f(C)$ for any closed curve $C$ persists. Indeed,
\[
i(\sigma_*(\mu_f),C)=i(\mu_f,\sigma_*(C))=i(\mu_f,C^{-1})=i(\mu_f,C).
\]
The first equality is by the fact that intersection number is invariant under the action of the flip map on both factors, and the fact that the flip map is an involution; the last equality is because of symmetry of intersection number.
Therefore, $\overline{\mu_f}$ is a geodesic current with the same intersection numbers as the oriented geodesic current $\mu_f$.

       Finally, the uniqueness of the geodesic current $\overline{\mu_f}$ is guaranteed by the fact that geodesic currents are uniquely determined by their intersection numbers with all closed curves by~\cite[Th\'eor\`eme~2]{Otal90:SpectreMarqueNegative}.
\end{proof}


\begin{remark}
The end of the proof of Proposition~\ref{prop:recoverintersection} is the only place where we have used the symmetry condition of $f$. In fact, note that before that step, the proof shows the curve functional induced by the intersection number must be symmetric. 
The symmetry condition has also been used in Proposition~\ref{prop:cont_conn_disc}, and will be used in proving the equivalent algebraic statement Theorem~\ref{thm:intersections_group}.
This condition is thus crucial to recover curve functionals as intersection numbers. In Section~\ref{sec:asymmetric} we show that the asymmetric intersection numbers do not satisfy an analogous duality result.
\end{remark}

\subsection{Proof of Theorem~\texorpdfstring{\ref{thm:intersections_group}}{muf}}
\label{subsec:algebraic_vs_top}
We finish this section by proving its algebraic version, namely, Theorem~\ref{thm:intersections_group}.


\begin{proof}[Proof of Theorem~\ref{thm:intersections_group}]
Let $\overline{f}(a) \coloneqq \frac{1}{2}\left(f(a) + f(A) \right)$. Note that since $f$ is additive, $\overline{f}$ is also additive.
First, extend $\overline{f}$ linearly to $\mathbb{R}_{\geq 0}$-formal linear combinations of $\Gamma$, i.e., to weighted multi-curves, by the additivity of $\overline{f}$:
\[
\overline{f}(s \cdot a + t \cdot b) = s\cdot \overline{f}(a) + t \cdot \overline{f}(b)
\]
for every $s,t \geq 0$, and $a,b \in \Gamma$.

To prove the theorem, we will show that (this extension of) $\overline{f}$ satisfies the hypothesis of Theorem~\ref{thm:intersect}, and then apply that theorem to conclude.

The only condition that requires verification is smoothing.

Start by assuming the essential crossing arises from an essential crossing between two curves $C_1$ and $C_2$ represented by elements $a,b$, and we will conclude disconnected smoothing.
In this case, either $(\ora{a},\ora{b})$ are $R$-crossing, or $(\ora{a},\ora{b})$ are $L$-crossing. If $(\ora{a},\ora{b})$ is $R$-crossing, hence oriented disconnected smoothing is satisfied by the hypothesis.  If $(\ora{a},\ora{b})$ is $L$-crossing, $(\ora{b},\ora{a})$ and, similarly, the hypothesis tells us
\[
\overline{f}(a) + \overline{f}(b) = \overline{f}(b) + \overline{f}(a) \geq \overline{f}(ba)= \overline{f}(ab)
\]
where just used in the last equality that $\overline{f}$ is conjugacy invariant.
Now, we assume an essential self-crossing of a multi-curve, and prove connected smoothing.

Using Proposition~\ref{prop:crossings-triality}, assume the essential self-crossing is realized by the self-intersection of a curve $[\gamma]=[ab]$, where $\ora{a},\ora{b}$ are parallel (the anti-parallel case is similar).
If $\ora{a},\ora{b}$ are $R$-parallel we have, by assumption, we have that oriented connected smoothing is satisfied.
If $\ora{b},\ora{a}$ are $L$-parallel, then $\ora{a},\ora{b}$ are $R$-parallel. Hence
\[
\overline{f}(ab)=\overline{f}(ba) \geq \overline{f}(b) + \overline{f}(a) = \overline{f}(a) + \overline{f}(b).
\]
If $(a,b)$ are $L$-parallel, then $(b,a)$ are $R$-parallel, thus
\[
\overline{f}(ab)=\overline{f}(ba)\geq \overline{f}(b) + \overline{f}(a) = \overline{f}(a) + \overline{f}(b). 
\]
Hence, oriented connected smoothing is satisfied in this case, too.
Finally, by Lemma~\ref{lem:oriented_smoothing_unoriented} $\overline{f}$ satisfies smoothing. We are thus under the hypotheses of Theorem~\ref{thm:intersect}.
\end{proof}
