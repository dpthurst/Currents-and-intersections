\section{Construction of the geodesic current}
\label{sec:construct}



\subsection{$\mu_{f}$ is countably additive}
\label{sec:countable-additivity}

We prove countable additivity of the function $\mu_{f}$ by
proving approximation results.



\begin{definition}
Let the right handed lift system $[p_1,q_1,p_2,q_2]$ determine a right-handed box $B = [a_1,a_2) \times [b_1,b_2)$.
We define a \emph{top vertical accordion} of boxes based at $B$ the sequence of boxes, indexed by $k$, given for each $k$, by
\[
p_1^{(k)}=p_1,p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)} \coloneqq q_1 x^k q_1^{-1}q_2.
\]
Since the endpoint corresponding to $q_2$ is the one changing, we will denote these sequence of boxes by
\[
F_{p_1,p_2,q_1,\vec{q_2}}.
\]
This defines boxes whose vertical direction becomes narrower as $k$ grows, with the top endpoint fixed. 
Similarly, we define \emph{bottom vertical accordion}, and denote it by 
\[
F_{p_1,p_2,\bar{q_1},q_2},
\]
giving a sequence of boxes whose vertical direction becomes narrower and the bottom endpoint fixed, i.e.,
\[
p_1^{(k)}=p_1,p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)} \coloneqq q_1 x^k q_1^{-1}q_2.
\]

On the other hand, if we fix $p_2,q_1,q_2$ and allow $p_1$ to vary, by choosing nested lifts of $x$ for $p_1$ and $p_2$, we obtain a datum given by the box notation $p_2^{(k)}=p_2,q_1^{(k)}=q_1,q_2^{(k)}=q_2$, and
\[
p_1^{(k)} = p_2 x^k p_2^{-1} p_1.
\]
and we denote the corresponding family of boxes,
\[
F_{\vec{p_1},p_2,q_1,q_2}.
\]
We will call them a \emph{right horizontal accordion} of boxes based at $B$ (the right endpoint is fixed).
Similarly, we can define a \emph{left horizontal accordion} (the left endpoint is fixed).
\label{def:accordion}
\end{definition}

The following Proposition will be key to control how much $\mu_f$ when the right-handed boxes are varied in a control way. In particular, it will be crucial in proving countable addivitity of $\mu_f$.

\begin{proposition}
 Given a rhs system $[p_1,p_2,q_1,q_2]$ so that, moreover, $q_1^{-1}q_2$ cross $x$ to the right. let $F(k)$ be  a system of lifts in an accordion configuration, as in Definition~\ref{def:accordion}, in any of the four types $F_{\vec{p_1},p_2,q_1,q_2}(k)$, $F_{p_1,p_2,q_1,\vec{q_2}}$, $F_{p_1,\vec{p_2},q_1,q_2}(k)$ and $F_{p_1,p_2,\vec{q_1},q_2}(k)$. We have

\begin{itemize}
 \item For every $k \geq 0$, $F(k)$ is a right-handed box.
\item For every $k$, $F(k+1) \subset F(k)$
\item
\[
\lim_{k \to \infty} \mu_f(F(k))= 0
\]
\end{itemize}
\label{prop:finalfamily1}
\end{proposition}
\begin{proof}
\begin{itemize}
    \item We first prove that an accordion $F(k)$ is a right-handed box. Consider first a top vertical accordion $F_{\vec{p_1},p_2,q_1,q_2}(k)$ (the other types are similar).
    We know that $[p_1,p_2,q_1,q_2]$ form a rhs system, so 
    $p_i q_i$ cross $x$ to the right
        and $q_2 x^n p_1$ cross $q_1x^n p_2$ to the right (by Definition~\ref{def:righthandedcrossing}). 
        By north-south dynamics of $q_1 x q_1^{-1}$, the translates of $\tilde{x}$ look like in Figure~\ref{fig:finalfamily}.
        The following properties follow immediately.
    
    \begin{enumerate}
        \item $p_1 q_1 x^k q_1^{-1} q_2$ crosses $x$ to the right.
        \item $p_2 q_1 x^k q_1^{-1} q_2$ crosses $x$ to the right.   
        \item   $q_1 x^k q_1^{-1} q_2x^n p_1$ cross $q_1x^n p_2$ to the right.
    \end{enumerate} 
    This proves that top vertical accordions are right handed boxes. The proof is similar for other types of accordions.
    \item We prove that for a fixed rhs sytem $[p_1,p_2,q_1,q_2]$ and a corresponding accordion $F(k)$, $\lim_{k \to \infty} \mu(F(k))=0$.
    We give the argument for type $F_{p_1,p_2,q_1,\vec{q_2}}$, the other types are similar.
    We will first go from the box notation to the orbit notation, since the latter is better suited for the definition of $\mu_f$.
    We have
 \begin{align*}
     a &= p_1 q_1 \\
     b &= p_1 q_1 x^k q_1^{-1} q_2 \\     
     c &= p_2 q_1  \\  
     d &= p_2 q_1 x^k q_1^{-1} q_2 \\  
 \end{align*}
 Writing $e=q_1^{-1} q_2$, we have
 \begin{align*}
     b &= a x^k e \\     
     d &= c x^k e \\  
 \end{align*}
We thus get the family of box orbits given by
\[
\bigl([B]_{a,a x^k e,c,c x^k e}\bigr)_{k=0}^\infty.
\]
Figure~\ref{fig:finalfamily} related the box to the orbit notation.

    We want to show
\[
\lim_{k \to \infty} \mu_f([B]_{a,a x^k e,c,c x^k e})=0
\]
By definition of $\mu_f$,
\begin{align*}
\mu_f([B]_{a,a x^k e,c,c x^k e})=\lim_{n\to \infty} f([ax^ke x^n]) + f([cx^n])-f([ax^n])-f([cx^kex^n]).
\end{align*}
So, we are left to evaluate the nested limit
\begin{align*}
\lim_{k \to \infty} \lim_{n\to \infty} f([ax^ke x^n]) + f([cx^n])-f([ax^n])-f([cx^kex^n]).
\end{align*}
    We use Proposition~\ref{prop:join-split-geom},
    equations~\ref{eq:cross-join} and
    \ref{eq:cross-split},
    \DPTtodo{Needs tweaking for revised statements}
    taking $r=1$ and $r'=2$ (using the fact that if $x$ is simple, we can take $s=0$ and $s'=1$).
    We then have the expression in the above iterated limit is upper bounded by
    \[
  f([ax^{k+1}]) + f([e x^{n-1}]) + f([cx^n])-f([ax^n])-f([cx^{k+2}]) - f([ex^{n-2}])
    \]
    by adding and subtracting a term $f([x^{k+n}])$, the iterated limit of the last expression is equivalent to the iterated limit
    \begin{align*}
       \lim_{k\to\infty} \lim_{n\to\infty}  \bigl( f([ax^{k+1}]) - f([x^{k+1}]\bigr) + \bigl( f([ex^{n-1}]) - f([x^{n-1}])\bigr) +  \\
        \bigl( f([cx^n]) - f([x^n]) \bigr) - \bigl(f([cx^n]) - f([x^n])\bigr)\\
        -\bigl(f([cx^{k+2}])-f([x^{k+2}])\bigr)-\bigl(f([ex^{n-2}])-f([x^{n-2}])\bigr)
    \end{align*}
    Finally, this iterated limit is 0 by  Lemma~\ref{lem:additivity-limit}, since $e=q_1^{-1} q_2$ crosses $x$ to the right, by assumption.\qedhere
    \end{itemize}
\end{proof}

We now use Proposition~\ref{prop:finalfamily1} to prove that $\mu_f$
is countably additive.

\begin{remark}
We remark that at this point of the argument, we could also use a more indirect strategy, using Riesz representation, as done in \cite[Proposition~3.11]{BIPP21:PositiveCrossratios}.
Indeed, the endpoints of r.h. boxes correspond to endpoints of lifts of $x$, and these are dense in $\partial S$ by Lemma~\ref{lem:minimal}. Moreover, if $R,R_1,R_2$ are rh boxes with $R=R_1 \cup R_2$, with $R_1,R_2$ disjoint, $\mu_f(R)=\mu_f(R_1)+\mu_f(R_2)$, by Proposition~\ref{prop:sidebyside}.
Also, by the argument in Proposition~\ref{prop:RHsemiring}, for $R,R'$ r.h. boxes, if $R \subset R'$, we have $\mu_f(R) \leq \mu_f(R')$.
Hence, our function $\mu_f$ satisfies the hypotheses of~\cite[Proposition~4.1]{BIPP21:PositiveCrossratios}, and hence by ~\cite[Proposition~4.3]{BIPP21:PositiveCrossratios} we obtain a geodesic current with desired properties.
We choose to give an independent argument that showcases the fact that r.h. boxes enjoy the extra structure of a semi-ring.
\end{remark}
We will need to consider more types of boxes
than the half-open boxes we have been working with so far. For $B$ a
right-handed box $[a_1,a_2) \times [b_1,b_2)$, let
$\overline{B} \coloneqq [a_1,a_2] \times [b_1,b_2]$ and
$B^\circ \coloneqq (a_1,a_2) \times (b_1,b_2)$ be the corresponding
closed and open boxes.

\begin{lemma}\label{lem:extend1}
  For $B = [a_1,a_2) \times [b_1,b_2)$ a right-handed box and
  any $\epsilon > 0$, there is a right-handed box $V(B,\epsilon) =
  [a_1',a_2) \times [b_1',b_2)$ in $\mathcal{R}$ so that
  $a_1' < a_1$ and $b_1' < b_1$ and
  $\mu_f(V(B,\epsilon) \setminus B) < \epsilon$.
\end{lemma}

That is, $V(B,\epsilon)$ extends $B$ slightly down and to the left,
without increasing the mass much.

\begin{proof}
  Let $B = [a_1,a_2) \times [b_1,b_2)$. By Proposition~\ref{prop:density} there is a right handed box $B'=[a_1'',a_1) \times [b_1'',b_1)$ in $\mathcal{R}$ so that
  $a_1'' < a_1$ and $b_1'' < b_1$.
  Let $H$ be a right horizontal accordion based at $B'$. \DMGtodo[inline]{what does based at mean?}
  By Proposition~\ref{prop:finalfamily1}, for $k$ large enough, we have $\mu_f(F(k))<\epsilon$. 
  Let $V'(B,\epsilon) \coloneqq B \cup F(k)$.
  We can now repeat the same procedure with $V'(B,\epsilon)$ in the vertical direction.
\end{proof}

\begin{lemma}\label{lem:extend2}
  For $B = [a_1,a_2) \times [b_1,b_2)$ a right-handed box and
  any $\epsilon > 0$, there is a region $W(B,\epsilon) =
  \bigl([a_1,a_2) \times [b_1',b_2)\bigr) \cup
  \bigl([a_2',a_2) \times [b_1,b_2)\bigr)$ in $\mathcal{R}$ so that
  $a_1 < a_2' < a_2$ and $b_1 < b_2' < b_2$ and
  $\mu_f(W(B,\epsilon)) < \epsilon$.
\end{lemma}

That is, $W(B,\epsilon)$ is a neighborhood of the upper and right
edges of~$B$ with small mass.

\begin{proof}
Let $H(k)$ be a right horizontal accordion, and $F(k)$ a top vertical accordion.
We tan take $k$ large enough so that $\mu_f(H(k)) < \epsilon/2$,  $\mu_f(F(k)) < \epsilon/2$, and then let $W(B,\epsilon) = H(k) \cup V(k)$.
\end{proof}
  
In the setting of Lemma~\ref{lem:extend2}, also set
\[
  W^\circ(B,\epsilon) \coloneqq \bigl([a_1,a_2] \times (b_2',b_2]\bigr) \cup
    \bigl((a_2',a_2] \times [b_1,b_2]\bigr);
\]
this is an open neighborhood of the upper-right edges of the
$\overline{B}$ in the subspace topology.

\begin{proposition}\label{prop:countablyadditive}
  The finitely additive measure $\mu_f$ is countably subadditive.
\end{proposition}

\begin{proof}
  Let $B_0$ be a right-handed box, and suppose we are given a
  countable covering of it: $B_0 \subset \bigcup_{i=1}^\infty B_i$. We
  must show that
  \[
    \mu_f(B_0) \le \sum_{i=1}^\infty \mu_f(B_i).
  \]
  Fix $\epsilon > 0$. For $i \ge 1$, set
  $V_i \coloneqq V(B_i, \epsilon/2^i)$. Also add one more set by setting
  $V_0 \coloneqq W(B_0,\epsilon)$ and $V_0^\circ \coloneqq W^\circ(B_0,\epsilon)$.

  Then $\{V_i^\circ\}_{i=0}^\infty$ cover $\overline{B_0}$, since
  the $B_i$ cover $B_0$, $B_i \subset V_i^\circ$, and
  $\overline{B_0}\setminus B_0 \subset V_0^\circ$. Since
  $\overline{B_0}$ is compact and the $V_i^\circ$ are open (in the
  subspace topology), we have a finite subcover; let $I \subset
  \mathbb{N}$ be the set of
  indices included in the subcover.

  By finite additivity, we then have
  \begin{align*}
    \mu_f(B_0) &\le \sum_{i \in I} \mu_f(V_i)\\
      &\le \sum_{i \in I, i \ne 0} \mu_f(B_i) + \sum_{i \in I, i \ne 0}
        \mu_f(V_i \setminus B_i) + \mu_f(V_0)\\
      &< \sum_{i=1}^\infty \mu_f(B_i) + 2\epsilon.
  \end{align*}
  Since $\epsilon$ was arbitrary, we are done.
\end{proof}

We now put everything together to show that $\mu_f$ is a geodesic current.

\begin{proposition}
Let $f \colon \mathcal{C}(S) \to \mathbb{R}$ be a functional on curves satisfying additivity, smoothing, stability and homogeneity.
Then $\mu_f$ is a geodesic current.
\end{proposition}
\begin{proof}
By Proposition~\ref{prop:limitexists},
Proposition~\ref{prop:RHsemiring}, and
Proposition~\ref{prop:countablyadditive}, $\mu_f$ is a finitely
additive and countably subbadditive pre-measure on $\mathcal{R}$. 
Therefore, by Carathéodory extension Theorem~\ref{thm:caratheodory}, $\mu_f$ can be extend to a measure $\mu_f$ on $\mathcal{A}_{\mathcal{R}}$, the $\sigma$-algebra generated by the semi-ring $\mathcal{R}$.
Let $\mathcal{B}$ be the Borel $\sigma$-algebra of $G(\wt{\Sigma})$.
Since $\mathcal{A} \subset \mathcal{B}$, we have $\mathcal{A}_{\mathcal{R}} \subset \mathcal{B}$.
On the other hand, given a half-open box of geodesics $B=[a,b) \times [c,d) \in \mathcal{B}$, it can be written as an intersection of two countable unions of boxes in $\mathcal{R}$, since, by Proposition~\ref{prop:density}, the set of elements in $\mathcal{R}$ are dense in the subset of half-open rectangles of $G(\wt{\Sigma})$ (this is the same as the standard argument one uses to prove the Borel $\sigma$-algebra generated by half-rectangles with rational endpoints is the usual Borel $\sigma$-algebra).
Note that $\mu_f$ is locally finite, since any compact subset of $G(\wt{\Sigma})$ can be covered by one of the boxes in $\mathcal{R}$, where the measure takes a finite value.
Therefore, $\mu_f$ is a Borel and locally finite measure. 
By definition of the pre-measure $\mu_f$, this measure $\mu_f$ is $\pi_1(S)$-invariant. Finally, $\mu_f$ is non-negative by Proposition~\ref{prop:box-positive}.
It remains to address flip invariance.
We can consider $\frac{1}{2}(\mu_f + \sigma_{*} \mu_f)$ to obtain a flip invariant measure, since if $\sigma_*$ is the flip map on oriented geodesic currents (Definition~\ref{def:unorientedcurrents}), we have, for all closed curve $C$,
\[
i(\sigma_*(\mu_f),C)=i(\mu_f,\sigma_*(C))=i(\mu_f,C^{-1})=i(\mu_f,C).
\]
The first equality is by the fact that intersection number is invariant under the action of the flip map on both factors, and the fact that the flip map is an involution; the last equality is because of symmetry of intersection number.
Therefore, $\frac{1}{2}(\mu_f + \sigma_{*} \mu_f)$ is a geodesic current.\qedhere
\end{proof}

\subsection{$\mu_{f}$ recovers $f$ on curves}
\label{sec:recovery}

It remains to show that $\mu_f$ recovers $f$ as an intersection number on curves.

If $y$ is a hyperbolic element representing a closed curve
$C \subset \Sigma$, let
$z \in \partial_{\infty} \wt{\Sigma} - \{ y_-, y_+\}$ so that
$y_-, z, y \cdot z, y_+$ are in cyclic order.
Then
$(y_+,y_-) \times [z,y \cdot z)$ is a Borel fundamental domain for the
$\langle y \rangle$-action on $\DG(\wt{\Sigma})$, and the intersection
of $C$ with a geodesic current $\mu$ can be computed as
\[
i(\mu, C) = \mu((y_+,y_-) \times (z,y \cdot z)).
\]
(see \cite[Lemma~4.4]{MZ19:PositivelyRatioed}, cf. \cite[Page~153]{Otal90:SpectreMarqueNegative} for details).

\begin{figure}
\centering{
\resizebox{80mm}{!}{\svgfig{fundamental}}
\caption{Fundamental domain}
\label{fig:fundamental}
}
\end{figure}


\begin{proposition}
For $C$ a closed curve, we have
\[
i(\mu_f,C)=f(C).
\]
\label{prop:recoverintersection}
\end{proposition}
\begin{proof}
  Given a representative $y \in \pi_1(S,*)$ of the conjugacy class~$C$, we will first show that we can use right handed boxes to approximate a fundamental domain for $\supp \mu \times \{ \tilde{y} \}$ in $\DG(\Sigma)$.
Second, we'll show that computing the measure of that fundamental domain yields $f([y])$.


Choose a conjugate $y$ in $[y]$, 
and $p_1, q_1 \in \pi_1(S,*)$, so that $x$ crosses $p_1$ to the right, i.e., $p_1^{-1}$ crosses $x$ to the right, and $p_1$ crosses $y$ to the right.
Furthermore, choose $q_1$ so that $q_1$ crosses $x$ and $y$ to the right. 
Then, by Lemma~\ref{lem:rhs}, we have a configuration of lifts like the upper left and right lifts of $\wt{x}$ in Figure~\ref{fig:fundamental}.
Moreover, this choice guarantees that $q_1 p_1$ crosses $y$ to the right and thus, by Lemma~\ref{lem:rhs}, $\wt{y}$ is parallel to $q_1 p_1 \cdot \wt{y}$.

For each $n$, define the following elements of $\pi_1$:
\begin{align*}
  p_1^n &\coloneqq p_1 & q_1^n &\coloneqq y^n q_1\\
  p_2^n &\coloneqq p_1 y &q_2^n &\coloneqq y^{-(n+1)} q_1.
\end{align*}
\begin{claim}
The datum $[p_1^n,p_2^n,q_1^n,q_2^n]$, as defined above, is a rhs system, i.e.,
\begin{enumerate}
    \item $p_i^nq_j^n$ cross $x$ to the right, for $i,j=1,2$.
    \item $q_2^n x^m p_1^n$ crosses $q_1^n x^m p_2^n$ to the right, for $m \geq 0$. 
\end{enumerate}
\end{claim}
\begin{proof}
By NS dynamics of $y$, the translates of $\wt{x}$ by $(p_i^{(n)})^{-1}$ and $q_i^{(n)}$, for $i=1,2$, are like in Figure~\ref{fig:fundamental}.
\end{proof}

If we translate the rhs system
$[p_1^{(n)},p_2^{(n)},q_1^{(n)},q_2^{(n)}]$ into the orbit notation,
and set $p \coloneqq p_1$ and $q \coloneqq q_1$, we have right-handed boxes
\begin{align*}
  a^{(n)} &= p y^{n}   q &  b^{(n)} &= p y^{-(n+1)} q \\
  c^{(n)} &= p y^{n+1} q &  d^{(n)} &= p y^{-n} q.
\end{align*}
From Lemma~\ref{lem:axbymidconvex}, we know $f(py^{n}qx^m)$ is $r$-convex in the variable $y$.
Let $w \coloneqq y^r$.
As a consequence, $f(pw^{n}qx^m)$ is mid convex in the variable $w$, i.e., 
\begin{equation}
f(pw^{n+1}qx^m) + f(pw^{n-1}qx^m) \geq 2f(pw^{n}qx^m).
\label{eq:midconvex_word}
\end{equation}
We define $f^{\pm}(n,m) \coloneqq f(pw^{\pm n}qx^m)$ and  $d_{n,m} \coloneqq f^+(n,m) + f^-(n,m)$.
Let $a_{n,m} = d_{n,m} - d_{n-1,m}$.
By Equation~\ref{eq:midconvex_word}, $d_{n,m}$ is mid convex in $n$.
This tells us that $a_{n,m}$ is increasing in $n$.
Note that, by definition of the measure $\mu_f$, we have
\[
\lim_m a_{n,m} = \mu_f(B_n)
\]
where $B_n \coloneqq B_{a^{(n)}, b^{(n)}, c^{(n)}, d^{(n)}}$.
Let $b_{n,m} \coloneqq a_{n,m} - a_{n-1,m}$.
We thus have
\[
\lim_m b_{n,m} = \mu_f(B_n) - \mu_f(B_{n-1}).
\]
Let $c_n \coloneqq \mu_f(B_n) - \mu_f(B_{n-1})$.
Since $B_{n-1} \subset B_n$, we have $c_n \geq 0$.
Moreover,
\[
\lim_{N \to \infty} \sum_{n=1}^N c_n = \lim_{N \to \infty} \mu_f(B_N) - \mu_f(B_1).
\]
This last limit exists since the sequence $\mu_f(B_N) - \mu_f(B_1)$ is weakly monotone increasing, and
by Proposition~\ref{prop:bounded_double}, it is bounded.
Hence, the assumptions of Proposition~\ref{prop:exchange_limits} apply to $a_{n,m}$ and thus we can exchange limits
\[
\lim_m \lim_n a_{n,m} = \lim_n \lim_m a_{n,m}
\]

Now, if we let $z=(p_1^{-1} \cdot \wt{x})^-$, and replace $y$ by $w=y^r$ in the above definition of $B_{a^{(n)}, b^{(n)}, c^{(n)}, d^{(n)}}$ we have
    \begin{multline*}
      \begin{aligned}
      \mu_f((\gamma_-,\gamma_+) \times [z,w \cdot z)) \\
    &= \lim_{n\to\infty} \mu_f(B_{a^{(n)}, b^{(n)}, c^{(n)}, d^{(n)}})  \\
    &= \frac{1}{2}\lim_{n\to\infty} \lim_{m\to\infty}f([pw^{n+1}qx^{m}]) + f([pw^{-(n+1)}qx^{m}])\\
    &-f([pw^{n}qx^{m}]) - f([pw^{-n}qx^{m}]) \\
    &= \frac{1}{2}\lim_{m\to\infty} \lim_{n\to\infty}f([pw^{n+1}qx^{m}]) -f([pw^{n}qx^{m}]) -  \\
        f([pw^{-(n+1)}qx^{m}]) - f([pw^{-n}qx^{m}]) \\
    &= \frac{1}{2}\lim_{m\to\infty} \lim_{n\to\infty} \bigl( f([pw^{n+1}qx^{m}]) - f([w^{n+1}])\bigr) \\
    &-\bigl( f([pw^{n}qx^{m}]) - f([w^{-(n+1)}])\bigr)+  \\
    &    \bigl( f([pw^{-(n+1)}qx^{m}]) - f([w^{n}])\bigr) \\
    &- \bigl( f([pw^{-n}qx^{m}]) -  f([w^{-n}])\bigr) +\\
    &    f([w]) + f([w^{-1}]) \\
    &= \frac{1}{2}(f([w]) + f([w^{-1}]))\\
    &=f([w])
      \end{aligned}
    \end{multline*}      The first equality follows by the fact that the boxes defined above converge from below to $(\gamma_-,\gamma_+) \times [z,y
      \cdot z)$ as $n \to \infty$. The second equality is by
      definition of the measure $\mu_f$ on boxes.
      For the third, we exchange nested limits thanks to the above. \footnote{This is a key step, since upon exchanging the limit, the nested limit can be evaluated. Note this was not possible before: it was possible to conclude the limit existed, but not what its value was.} For the fourth we add and subtract $f([w^{n+1}])$, as well as $f([w^{-(n+1)}])$ and collect terms in brackets.
     By Lemma~\ref{lem:axbymidconvex} and~Proposition~\ref{prop:bounded_double}, they are mid-convex bounded functions of $n$.
     Therefore, by Lemma~\ref{lem:limitconvex},  the bracketed terms have well-defined limits in $n$. It is key to notice that these limits come in pairs of opposite signs, and thus cancel out.
       In the sixth equality, we use the symmetry property of~$f$ to obtain the result.
\end{proof}

\begin{remark}
The proof of Proposition~\ref{prop:recoverintersection} is the only place where we have used the symmetry condition of $f$. This condition is crucial to recover curve functionals as intersection numbers, as we explain in Section~\ref{sec:asymmetric}.
\end{remark}
