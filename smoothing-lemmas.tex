\subsection{Smoothing property and existence of limits}


Now we use the lemmas obtained in the previous subsection on our function on curves $f$ satisfying smoothing, convex union, homogeneity and stability to prove the existence of certain limits that will be crucial to construct our measure.

An immediate consequence of smoothing and convex union is positivity of $f$.

\begin{corollary}
  Let $f$ be a functional satisfying convex union and smoothing. Then $f([a]) \geq 0$ for any $a \in \pi_1(S)$.
  \label{cor:positivity}
\end{corollary}
\begin{proof}
Let $x \in \pi_1(S)$ so that $a$ crosses $x$ to the right. By oriented smoothing,
\[
[a][x] \reducesto [ax].
\]
On the other hand, by Equation~\ref{eq:cross-red3} (which uses unoriented smoothing), we have
\[
[ax][a]\reducesto [x]
\]
Since $f$ satisfies smoothing (oriented and unoriented) as well as convex union, we get
\begin{align*}
    f(a) + f(x) &\geq f([a] [x]) \geq f(ax)\\
     f(ax) + f(a) &\geq f([ax] [a]) \geq f(x).\\   
\end{align*}
Adding both equations, we get $f(a) \geq 0$, as we wanted.
\end{proof}

A similar argument with different assumptions also yields positivity.

\begin{corollary}
  Let $f$ be a functional satisfying convex union and quasi-smoothing and stability/homogeneity. Then $f([a]) \geq 0$ for any $a \in \pi_1(S)$.
  \label{cor:positivity2}
\end{corollary}

\begin{proof}
Let $x \in \pi_1(S)$ so that $a$ crosses $x$ to the right. As in the proof of Corollary~\ref{cor:positivity}, we have
\[
[a^n][x] \reducesto [a^nx].
\]
and
\[
[a^nx][a^n]\reducesto [x]
\]
Since $f$ satisfies quasi-smoothing (oriented and unoriented) as well as convex union, we get
\begin{align*}
    f(a^n) + f(x) &\geq f([a^n] [x]) \geq f(a^nx) - R\\
     f(a^nx) + f(a^n) &\geq f([ax] [a^n]) \geq f(x) - R.\\   
\end{align*}
Adding both equations, we get $f(a^n) \geq -2R$. By stability and homogeneity, this yields $n f(a) \geq -2R$, and taking $n$ large enough we see that $f(a) \geq 0$.
\end{proof}


The following lemma follows by the definition of right-handed crossing (Definition~\ref{def:righthandedcrossing}).
\begin{lemma}
Let $a,x \in \pi_1(\Sigma)$ so that $a$ crosses $x$ to the right.
\[
[a][x^n] \reducesto [ax^n], \mbox{ for all } n \in \mathbb{N}
\]
\label{lem:P}
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:rhs}, we have that
 \[
 [ax^m][x^n] \reducesto [ax^{n+m}]
 \]
 which for $m=0$ yields
 \[
 [a][x^n] \reducesto [ax^{n}].
 \] 
\end{proof}

We will also need the following converse smoothing.

\begin{lemma}
Let $a,x \in \pi_1(\Sigma)$ so that $a$ crosses $x$ to the right. There exists $s_0=s_0(a) \in \mathbb{N}$ so that
\[
[ax^n][x] \reducesto [x], \mbox{ for all } n \geq s_0,
\]
\label{lem:Q}
\end{lemma}
\begin{proof}
The claim follows by Equation~\ref{eq:cross-red3} in Proposition~\ref{prop:join-split-geom}, taking $m=0$.
\end{proof}

The following result shows a key limit to construct the measure exists and uses the smoothing lemmas.

\begin{corollary}\label{cor:additivity-limit}
Let $f$ be a curve functional satisfying convex union and smoothing. Let $a,x \in \pi_1(\Sigma)$.
  If $a$ crosses $x$ to the right, then
  \[
    \lim_{n \to \infty} f([a x^n]) - f([x^n])
  \]
  exists.
\end{corollary}

\begin{proof}
  By Eq.~\eqref{eq:cross-red2}, with $a=b$, there exists $r>0$ so that, for all $n$,
  \[
     [ax^{n+r}] [ax^{n-r}] \reducesto [(ax^n)^2]
  \]
  (if $x$ is simple, we can take $r=1$).
  By smoothing and convex union, we get%
  \footnote{This is where smoothing as opposed to just
    quasi-smoothing is crucial. See Remark~\ref{rmk:nonconvex} for an example.}
  \[
  f([ax^{n-r}]) + f([ax^{n+r}]) \geq 2 f([ax^n]).
  \]
  Hence, $f([a x^n])$ is a $r$-midpoint convex function
  of $n$, and thus so is $g(n) \coloneqq f([a x^n]) - f([x^n])$.
 
 By Lemma~\ref{lem:P}, and by additivity and (oriented) smoothing properties of $f$, we get
 \[
 f([a]) + f([x^{n}]) \geq f([ax^{n}]).
 \]
By Lemma~\ref{lem:Q}, and by additivity and (unoriented) smoothing of $f$, we get, for
  \[
    -f([a]) \le f([a x^n]) - f([x^n]).
  \]
  This shows that $g$ is an $r$-midpoint convex and bounded function, so by Lemma~\ref{lem:limexistrmidconvex}, the limit in the statement exists.
\end{proof}

\begin{remark}\DMGtodo[inline]{(Punctures) This example uses a punctured surface. Extending our result to punctured case would make it more reasonable.}
\label{rmk:nonconvex}
The limit in Corollary~\ref{cor:additivity-limit} need not exist if one only assumes $f$ satisfies quasi-smoothing.
Consider the once-punctured torus with generating set $\{a,b,a^2\}$, where $a$ and $b$ are the standard generators of $\pi_1(S)$.
Let $f$ be the stable word-length with respect to this generating set.
By picking the right orientations, $a$ crosses $b$ right-handedly, thus $ab$ also crosses $b$ right-handedly by Lemma~~\ref{lem:rhs_prod}.
$f$ satisfies quasi-smoothing instead of smoothing (see \cite[Example~4.10]{MGT19:CurvesToCurrents}) for details). 
The expression in the limit in Corollary~\ref{cor:additivity-limit} for the pair $ab$ and $b$ is
\begin{equation*}
  f(ab^{n+1})-f(b^n)=\begin{cases} 1+ \frac{n+2}{2} - \frac{n}{2}=2& \mbox{ if } n \mbox{ is even } \\
    1+ \frac{n+1}{2} - \frac{n+1}{2}=1 & \mbox{ if } n \mbox{ is odd }
  \end{cases}
\end{equation*}
so the limit when $n$ goes to infinity doesn't exist.
\end{remark}

\begin{proof}[Proof of Proposition~\ref{prop:limitexists}]
This follows immediately from Corollary~\ref{cor:additivity-limit}.
\end{proof}

\begin{remark}
Convexity of $f([ax^n])$ as a function of $n$ also plays a key role in the work of Feng Luo of characterization of intersection numbers for simple closed curves (see \cite[Corollary~4.4]{Luo98:IntersectionNumbers}).
Feng Luo's work characterizes functions $f \colon \mathcal{S}(S) \to \mathbb{R}$ on simple closed curves that come from intersection numbers in terms of an infinite family of polynomial equations involving the $f$-value of triples of simple curves in the max-plus semi-ring structure on the real numbers.
One of the tools of this characterization is the fact that, for $a,b$ simple curves, the function $h(n) \coloneqq f([a^n b])$ is a convex.
In the proof of Corollary~\ref{cor:additivity-limit}, we generalize this to the case where $a,b$ might be non-simple, obtaining  $r$-midpoint convexity property of $g$.
Also, see Proposition~\ref{prop:feng} to see how our conditions on a curve function $f$ imply Feng Luo's conditions for the restriction of $f$ to simple curves.
\end{remark}

 Here we show that all the necessary hypotheses are satisfied to apply Theorem~\ref{thm:caratheodory}.

\begin{proposition}\label{prop:join-split-alg}
Let $a,b,x$ be arbitrary elements of $\pi_1(S)$.
  If $a$ and $b$ cross $x$ to the right, then the limits
  \[
    \lim_{n^{\pm},m^{\pm}} f([a x^n b x^m]) - f([ax^n]) - f([bx^m])
  \]
  exist and are equal to~$0$.
\end{proposition}
\begin{proof}
We separate the proof in two cases, depending on the magnitude of $|m-n|$.
\begin{itemize}
    \item Case 1: $|n-m|$ is large.  By Proposition~\ref{prop:join-split-geom}, we have
  \[
    f([a x^n]) + f([b x^{m}]) \ge f([a x^{n+1} b x^{m-1}])
      \ge f([a x^{n+2}]) + f([b x^{m-2}]).
  \]
  for all $m,n$ so that $m-n -s >r > s$, or $m-n+s<r<-s$, i.e.
  For the first one, it suffices to have $m-n-s > s+1$, i.e., $m-n>2s+1$
  For the second, $m-n+s<-s-1$, i.e. $m-n<-2s-1$.
  In any case, $|m-n|>2s+1$
  Thus, by Corollary~\ref{cor:additivity-limit}, the claim follows in this case.
  \item Case 2: $|m-n|$ is small.  Let $p=m-n$ and suppose that $|p| \leq 2s+1$. By Lemma~\ref{lem:rhs_prod}  $ax^nb$ crosses $x$ to the right. Therefore, we have that for every $k$
  \[
  [ax^nbx^{n+p}] [x^k] \reducesto [ax^nbx^{n+p+k}].
  \]
  On the other hand, $bx^{n+p+k}a$ crosses $x$ to the right, also, and since
  \[
  [ax^{n+k}bx^{n+p+k}]=[x^{n+k}bx^{n+p+k}a]=[bx^{n+p+k}ax^{n+k}]
  \]
  we have
  \[
  [bx^{n+p}ax^{n}] [x^k] \reducesto [bx^{n+p}ax^{n+k}].
  \]  
  Therefore, by applying $f$ and using smoothing and additivity, we have that
  for all $k, n$
  \begin{align*}
  [ax^{n+p} bx^{n-k}] 2[x^k] & \reducesto  [ax^{n+p} bx^{n}] [x^k]  \\
  &\reducesto  [ax^{n+p} bx^{n+k}]
  \end{align*}
  Subtracting $[x^{p+2n+k}]$, applying $f$ using smoothing and additivity, and taking the limit when $n,k$ go to infinity we see, by Case 1, that
  \[
  \lim_{n \to \infty} f([ax^{n+p}bx^n])-f([x^n])=0
  \]  
  as we wanted to show.\qedhere
\end{itemize}
 \end{proof}
 
We will also need to show existence of limits of words of the type $by^m ax^n$.

 \begin{lemma}
 Let $a, b, x, y \in \pi_1(\Sigma)$ configured as in Lemma~\ref{lem:initial}.
 Let $f \colon \Curves(S) \to \mathbb{R}$ satisfy convex union, smoothing, stability and homogeneity.
 Consider
 \[
 g(n,m) \coloneqq f(b y^m a x^n).
 \]
 Then, there exist $r_0,s_0 \in \mathbb{N}$ so that, 
 \begin{enumerate}
 \item For all $m$, and all $r>r_0$, $g(\cdot,m) \colon \mathbb{Z} \to \mathbb{R}$ is $r$-midpoint convex.
 \item For all $n$, and all $s>s_0$, $g(n,\cdot) \colon \mathbb{Z} \to \mathbb{R}$ is $s$-midpoint convex.
 \end{enumerate}
 \label{lem:axbymidconvex}
 \end{lemma}
\begin{proof}
It follows by applying $f$ to the left hand side of the smoothing resolutions in Proposition~\ref{prop:fundamental-aybx}, and using convex union, smoothing, stability and homogeneity in this order.
\end{proof}


 \begin{proposition}
  Let $a, b, x, y \in \pi_1(\Sigma)$ configured as in Lemma~\ref{lem:initial}.
 Let $f \colon \Curves(S) \to \mathbb{R}$ satisfy convex union, smoothing, stability and homogeneity.
 Then the double limits
 \[
\lim_{n^{\pm},m^{\pm}} f(b y^m a x^n) - mf(y) - nf(x)
\]
exist. (Here, $n^\pm$ means the limit for either $n \to \infty$ or $n
\to -\infty$, and similarly for $m^\pm$.)
\label{prop:doublelimitxy}
 \end{proposition}
 \begin{proof}
 We define $g(n,m) \coloneqq f(b y^m a x^n) - mf(y) - nf(x)$. We note that, by Lemma~\ref{lem:axbymidconvex}, $g$ is $(r,s)$-midconvex for some $r,s \in \mathbb{N}$.
 If we can prove that $g$ is bounded, by Lemma~\ref{lem:limexistrmidconvextwovar}, the limits will exist.
 On the one hand, 
 \begin{align*}
     g(n,m)&=f([ax^n by^m])-nf(x) - mf(y) \\
     &\leq f([ax^n b]) - n f([x]) \\
     &\leq f([ab])
 \end{align*}
 In the first inequality, we have used Lemma~\ref{lem:P} with $ax^nb$ in place of $a$, and $y$ in place of $x$. This is justified since $ax^nb$ crosses $y$ to the right. Indeed, by the initial configuration in Figure~\ref{fig:initial}, $b^{-1} \cdot \wt{x}$ is parallel to $a \cdot \wt{x}$, and thus, by Lemma~\ref{lem:cornering}, for all $n \in \mathbb{Z}$, $\wt{ax^nb}$ has one endpoint in the window $w(a^{-1})$ and the other endpoint in the window $w(b)$. Since $b^{-1} \cdot \wt{x}$ is parallel to $\wt{y}$ and $\wt{y}$ is parallel to $a \cdot \wt{x}$, according to the initial configuration in Figure~\ref{fig:initial}, it follows that $ax^nb$ crosses $y$ to the right.
 In the second inequality, we use Lemma~\ref{lem:P}, with $ba$ in place of $a$.  
 It is justified since, according to Figure~\ref{fig:initial}, $b^{-1}\cdot \wt{x}$ is parallel to $a \cdot \wt{x}$, so $\wt{x}$ is parallel to $ba \cdot \wt{x}$, and thus $ba$ crosses $x$ to the right.
 
 On the other hand, 
 \begin{align*}
     g(n,m)&=f([ax^n by^m])-nf(x) - mf(y) \\
     &\geq -f([ax^nb]) - n f([x]) \\
     &\geq -f([ba])
 \end{align*}
 In the first inequality, we have used Lemma~\ref{lem:Q} with $a x^n b$ in place of $a$ and $y$ in place of $x$, since, as shown above, $ax^nb$ crosses $y$ to the right.

 In the second inequality, we use Lemma~\ref{lem:P}, with $ba$ in place of $a$, since $ba$ crosses $x$ to the right, as shown above.  
 It follows that $g$ is bounded, and thus the limit exists.
 \end{proof}
