% DOI printing: keep the hyperlink, print in roman, and let the
% DOI break at '.', '/' and '-' (url's \UrlBreaks) instead of never.
\newcommand*{\doi}[1]{\href{https://doi.org/#1}{\begingroup\urlstyle{rm}\nolinkurl{doi:#1}\endgroup}}

\def\mathcenter#1{%
  \vcenter{\hbox{$#1$}}%
}

\def\graph#1{
        \includegraphics{#1}
}

\def\mathgraph#1{
        \mathcenter{\graph{#1}}
}

\def\mfigb#1{
        \mathcenter{\includegraphics[trim=-1 -1 -1 -1]{#1}}
}
